// =============================================================================
// Big target v13 - other players look different in YOUR client only (nobody else sees it, the server is not told):
//   clan mates      a small bull   (the npc Tiny Baby Buffalo 16019)
//   clan wars       golden pigs    (the npc Golden Pig 13035)
//   your party      never changed (the bot's party list, looked at every second)
//   your target     unchanged; !target - another model (transformation 305 Zaken) + a bigger body (effect 17)
//
// Two ways to change a player:
//   tr   the server describes every player in CharInfo (31); near its end stand the transformation he is in (the
//        model the client draws instead of him) and his visual effects. We hand the client a copy of his last
//        CharInfo with another transformation / more effects, as if the server sent it. Only the ~190 models of
//        the client's transformation list (Library\PvP\models.txt) - no golden pig, no small bull there.
//   npc  he is taken off the screen (DeleteObject 08) and an npc (NpcInfo 0C) is put there with HIS object id: any
//        npc of the client, with HIS name (so /target Name macros find it; !names off - the npc's own name and his
//        name as the title under it) and a blank title (one space: an empty one makes the client show the npc's
//        own title from its list - "80 lvl"). The server's moves, HP, skills of him go to the npc (the same id); clicking
//        it, the bot's binds (Engine.SetTarget) select / attack him as always. A new CharInfo of him from the
//        server (gear, buffs) -> the npc is put there again at once.
//        The NpcInfo is the one this server sends for monsters (protocol 196, rebuilt byte for byte from 150
//        sniffed packets); its move speed = his run speed / NPC_BASE (!speed tunes it). The name: mask bit 0x10 of
//        the first mask byte, the string right after the follow info (the Mobius order - every other field of it
//        matched the sniffed packets; no named npc was sniffed here yet).
// He was our target and was taken off -> selected again in the client only (MyTargetSelected B9, the server is not
// asked: a cast is not broken).
//
// Who is who: the clan - the bot's list (ClanID = ours); the wars - the server's RelationChanged (CE) about him:
// the bits 0x4000 (one-sided war) / 0x8000 (mutual war).
//
// A player whose CharInfo did not come since the start (around before it) cannot be changed: walk away and back.
//
// The end of CharInfo on this server (C = where the effect count stands, found from the end - see ListAt):
//   C-34 clan reputation | C-30 the transformation | C-26 agathion | C-22 1 byte |
//   C-21 curCp maxHp curHp maxMp curMp | C-1 1 byte | C the effect count (4 bytes) | the effects (2 bytes each) | 3 bytes
// From the end of his name (E): E+121 mAtkSpd, E+125 pAtkSpd, E+129 run speed (2 bytes), E+145 move multiplier
// (a double) - checked on 220 packets.
//
// Commands in the PARTY chat (#!clan off ...; only you see the answers):
//   !clan npc 16019 | !clan tr 103 | !clan off      the clan mates' look
//   !war npc 13035  | !war tr 104  | !war off       the clan wars' look
//   !target         the target's look on / off (off at the start);  !tr N  its model (!tr - a list, !tr 0 - own)
//   !big !head !both !ave N M !ave 0                 the target's effects
//   !names          the npcs with the players' names (on) / the npcs' own names, the player's as the title
//   !speed N        the npcs' base speed (default 120): an npc runs behind the player - less; ahead - more
//   !who            how many are changed now
//   !off / !on      everyone back as the server sent him / on again
//   !sniff          every packet into the log;  !dump  the target's CharInfo into the log
// Log: Library\Logs\big_target_<time>.log
// =============================================================================

const
  RING_N      = 64;         // server packets kept between two looks of the main loop
  RING_B      = 1024;       // bytes of a packet kept (a CharInfo is ~340-400)
  CRING_N     = 32;         // client packets kept (the sniff only)
  CRING_B     = 256;
  SNIFF_B     = 400;        // bytes of any other packet kept for the sniff log
  CACHE_N     = 160;        // players whose last CharInfo is kept
  REL_N       = 128;        // relations of players not kept yet
  TICK_MS     = 30;
  PASS_MS     = 1000;       // the clan of new players looked up again this often
  LOG_MAX     = 6000;       // lines in the log at most
  SAVE_MS     = 2000;
  MAX_AVE     = 64;
  TAIL_B      = 3;          // bytes after the effect list: true hero, hair accessory, ability points
  TR_BACK     = 30;         // the transformation stands this many bytes before the effect count
  LK_NONE     = 0;          // a look: his own
  LK_TR       = 1;          //   our CharInfo copy (transformation / effects)
  LK_NPC      = 2;          //   an npc in his place
  WAR_BITS    = $C000;      // RelationChanged: one-sided war 0x4000, mutual war 0x8000

function RdB(Data: Pointer; Off: integer): integer;
var P: Pointer;
begin
  P := Data;
  Inc(Cardinal(P), Off);
  result := PByte(P)^;
end;

var
  // the server ring: the packet hook writes numbers only, the main loop does the rest
  RgSz, RgLen, RgId1, RgId2: array[0..63] of integer;
  RgByte: array[0..65535] of integer;       // packet i: RgByte[i * RING_B + b]
  RgHead, RgTail, RgLost: integer;
  // the client ring (the sniff)
  CgSz, CgLen, CgId1, CgId2: array[0..31] of integer;
  CgByte: array[0..8191] of integer;
  CgHead, CgTail, CgLost: integer;
  // the players: their last CharInfo (without its opcode) as the server sent it
  CaOID, CaLen: array[0..159] of integer;
  CaByte: array[0..163839] of integer;      // player k: CaByte[k * RING_B + b]
  CaAt: array[0..159] of cardinal;
  CaGone: array[0..159] of boolean;          // the server took him off the screen (out of sight)
  CaKind: array[0..159] of integer;          // what the client shows of him now: LK_NONE / LK_TR / LK_NPC
  CaReq, CaSig: array[0..159] of integer;    // the look asked last (kind, its signature); -1 = ask again
  CaClan: array[0..159] of integer;          // his clan id (the bot's list), -1 not known yet
  CaRel: array[0..159] of integer;           // his relation bits (RelationChanged)
  RlOID, RlVal: array[0..127] of integer;    // relations of players not kept yet
  RlNext: integer;
  // the settings
  ClanKind, ClanId, WarKind, WarId: integer;
  TgOn: boolean;
  TgTr: integer;                             // the target's model (transformation); 0 = his own
  AveIDs: array[0..15] of integer;           // the target's effects
  AveN: integer;
  NpcBase: integer;
  NamesOn: boolean;                          // the npc gets the player's name (else the title)
  IsOn, SniffOn: boolean;
  MyOID, MyClan: integer;
  PtOID: array[0..15] of integer;            // the party (the bot's list)
  PtN: integer;
  PtRead: array[0..15] of integer;           // ReadParty: the list read now
  CurOID: integer;                           // the target as the script knows it (0 none)
  GuardUntil: cardinal;                      // our own reselect of the target: target changes ignored till then
  ToldOID: integer;                          // "no CharInfo of him yet" said for this one
  BadN: integer;
  PX, PY, PZ, PClan: integer;                // LivePos
  WK, WS: integer;                           // WantOf: the kind, the signature ...
  WA: boolean;                               // ... with the target's effects
  // the log
  LogL: TStringList;
  LogPath: string;
  LogDirty, LogFull: boolean;
  LostSaid: integer;

procedure Lg(S: string);
begin
  if LogL.Count >= LOG_MAX then begin
    if not LogFull then begin
      LogFull := true;
      LogL.Add('... the log is full (' + IntToStr(LOG_MAX) + ' lines) - restart the script for a new one');
      LogDirty := true;
    end;
    exit;
  end;
  LogL.Add(FormatDateTime('hh:nn:ss.zzz', Now) + ' ' + S);
  LogDirty := true;
end;

procedure Say(S: string);
begin
  ChatSay(S);
  Lg('SAY ' + S);
end;

// ===================== the hooks: numbers only =====================
// (strings made in the packet thread while the main loop makes its own crash the script)

procedure OnPacket(ID1, ID2: Cardinal; Data: Pointer; Size: Word);
var i, n, b: integer;
begin
  // CharInfo, DeleteObject, RelationChanged; everything with the sniff
  if (ID1 <> $31) and (ID1 <> $08) and (ID1 <> $CE) and (not SniffOn) then exit;
  if RgHead - RgTail >= RING_N then begin    // the main loop is behind: this one is lost
    RgLost := RgLost + 1;
    exit;
  end;
  i := RgHead mod RING_N;
  RgId1[i] := ID1;
  RgId2[i] := ID2;
  RgSz[i] := Size;
  n := Size;
  if (ID1 <> $31) and (n > SNIFF_B) then n := SNIFF_B;
  if n > RING_B then n := RING_B;
  for b := 0 to n - 1 do RgByte[i * RING_B + b] := RdB(Data, b);
  RgLen[i] := n;
  RgHead := RgHead + 1;
end;

// the party chat commands: the chat packet copied as numbers; ChatMenuPoll in the main loop reads it, sends it on
procedure OnCliPacket(ID1, ID2: Cardinal; Data: Pointer; Size: Word);
var i, n, b: integer;
begin
  if ID1 = $49 then ChatMenuOnCliPacketRaw(ID1, ID2, Data, Size);
  if not SniffOn then exit;
  if CgHead - CgTail >= CRING_N then begin
    CgLost := CgLost + 1;
    exit;
  end;
  i := CgHead mod CRING_N;
  CgId1[i] := ID1;
  CgId2[i] := ID2;
  CgSz[i] := Size;
  n := Size;
  if n > CRING_B then n := CRING_B;
  for b := 0 to n - 1 do CgByte[i * CRING_B + b] := RdB(Data, b);
  CgLen[i] := n;
  CgHead := CgHead + 1;
end;

// ===================== numbers and hex =====================

function CB(k, b: integer): integer;
begin
  result := CaByte[k * RING_B + b];
end;

function CI(k, b: integer): integer;
begin
  result := CB(k, b) or (CB(k, b + 1) shl 8) or (CB(k, b + 2) shl 16) or (CB(k, b + 3) shl 24);
end;

function CW(k, b: integer): integer;
begin
  result := CB(k, b) or (CB(k, b + 1) shl 8);
end;

// a positive double of player k at byte b (0 when it is not a plain positive number)
function CD(k, b: integer): double;
var Hi, E, j: integer;
  V: double;
begin
  result := 0;
  Hi := CI(k, b + 4);
  if Hi <= 0 then exit;
  E := ((Hi shr 20) and $7FF) - 1023;
  if (E < -10) or (E > 10) then exit;
  V := 1 + (Hi and $FFFFF) / 1048576.0;
  if E > 0 then for j := 1 to E do V := V * 2;
  if E < 0 then for j := 1 to -E do V := V / 2;
  result := V;
end;

function HexW(V: integer): string;
begin
  result := IntToHex(V and $FF, 2) + IntToHex((V shr 8) and $FF, 2);
end;

function HexI(V: integer): string;
begin
  result := HexW(V) + HexW(V shr 16);
end;

// a positive number as a 4-byte float (IEEE 754), hex
function HexF(V: double): string;
var E, M: integer;
begin
  if V <= 0 then V := 1;
  E := 0;
  while V >= 2 do begin
    V := V / 2;
    E := E + 1;
  end;
  while V < 1 do begin
    V := V * 2;
    E := E - 1;
  end;
  M := Round((V - 1) * 8388608);
  if M >= 8388608 then begin
    M := 0;
    E := E + 1;
  end;
  result := HexI(((E + 127) shl 23) or M);
end;

// bytes A..B-1 of player k as hex
function HexPart(k, A, B: integer): string;
var j: integer;
begin
  result := '';
  for j := A to B - 1 do result := result + IntToHex(CB(k, j), 2);
end;

// where the effect count of player k stands: from the end - the TAIL_B last bytes (true hero 0 / 100, hair
// accessory 0 / 1, ...), before them N effects (1..30000 each), before them the count N; right before it 1 byte and
// the 5 CP / HP / MP numbers (each 0..1000000). Exactly one such place -> it; else -1 (the packet is left alone)
function ListAt(k: integer): integer;
var N, C, j, V, Found, L: integer;
  Ok: boolean;
begin
  result := -1;
  Found := 0;
  L := CaLen[k];
  if L < 60 then exit;
  if (CB(k, L - 3) <> 0) and (CB(k, L - 3) <> 100) then exit;
  if CB(k, L - 2) > 1 then exit;
  for N := 0 to MAX_AVE do begin
    C := CaLen[k] - TAIL_B - N * 2 - 4;
    if C < 40 then break;
    if CI(k, C) <> N then continue;
    Ok := true;
    for j := 0 to N - 1 do begin
      V := CW(k, C + 4 + j * 2);
      if (V < 1) or (V > 30000) then Ok := false;
    end;
    if not Ok then continue;
    for j := 0 to 4 do begin
      V := CI(k, C - 21 + j * 4);
      if (V < 0) or (V > 1000000) then Ok := false;
    end;
    if not Ok then continue;
    Found := Found + 1;
    result := C;
  end;
  if Found <> 1 then result := -1;
end;

// the byte right after his name (the name: UTF-16 from byte 21, ended by two zero bytes)
function NameEnd(k: integer): integer;
var j: integer;
begin
  j := 21;
  while (j + 1 < CaLen[k]) and (CW(k, j) <> 0) do j := j + 2;
  result := j + 2;
end;

// ===================== the players =====================

function CacheIdx(OID: integer): integer;
var k: integer;
begin
  result := -1;
  if OID = 0 then exit;
  for k := 0 to CACHE_N - 1 do
    if CaOID[k] = OID then begin
      result := k;
      exit;
    end;
end;

// the slot for a player: his own, a free one, else the one not seen for the longest (never the target)
function CacheSlot(OID: integer): integer;
var k: integer;
  Old: cardinal;
begin
  result := CacheIdx(OID);
  if result >= 0 then exit;
  result := 0;
  Old := $FFFFFFFF;
  for k := 0 to CACHE_N - 1 do begin
    if CaOID[k] = 0 then begin
      result := k;
      exit;
    end;
    if (CaAt[k] < Old) and (CaOID[k] <> CurOID) then begin
      Old := CaAt[k];
      result := k;
    end;
  end;
end;

// the player OID in the bot's list -> PX PY PZ (where he is now), PClan; false: not in the list
function LivePos(OID: integer): boolean;
var i: integer;
begin
  result := false;
  for i := 0 to CharList.Count - 1 do if i < CharList.Count then
    if CharList(i).OID = OID then begin
      PX := CharList(i).X;
      PY := CharList(i).Y;
      PZ := CharList(i).Z;
      PClan := CharList(i).ClanID;
      result := true;
      exit;
    end;
end;

// the relation bits of OID the server told (0: none / not told)
function RelGet(OID: integer): integer;
var j: integer;
begin
  result := 0;
  for j := 0 to REL_N - 1 do
    if RlOID[j] = OID then begin
      result := RlVal[j];
      exit;
    end;
end;

procedure RelPut(OID, V: integer);
var j: integer;
begin
  for j := 0 to REL_N - 1 do
    if RlOID[j] = OID then begin
      RlVal[j] := V;
      exit;
    end;
  RlOID[RlNext] := OID;
  RlVal[RlNext] := V;
  RlNext := (RlNext + 1) mod REL_N;
end;

// the first 13 bytes of player k's CharInfo (a byte, x y z) with his position now when the bot knows it
function HeadHex(k: integer): string;
begin
  if LivePos(CaOID[k]) then result := HexPart(k, 0, 1) + HexI(PX) + HexI(PY) + HexI(PZ)
  else result := HexPart(k, 0, 13);
end;

// his own CharInfo (the position made now)
function OwnHex(k: integer): string;
begin
  result := HeadHex(k) + HexPart(k, 13, CaLen[k]);
end;

// his CharInfo with transformation Tr (0: his own) and, WithAve, the target's effects added; '' = the list not
// found / nothing differs from his own
function TrHex(k, Tr: integer; WithAve: boolean): string;
var C, N, i, j, Add, Was, Want: integer;
  Extra: string;
  Have: boolean;
begin
  result := '';
  C := ListAt(k);
  if C < 0 then exit;
  N := CI(k, C);
  Extra := '';
  Add := 0;
  if WithAve then
    for i := 0 to AveN - 1 do begin
      Have := false;
      for j := 0 to N - 1 do
        if CW(k, C + 4 + j * 2) = AveIDs[i] then Have := true;
      if Have then continue;
      Extra := Extra + HexW(AveIDs[i]);
      Add := Add + 1;
    end;
  Was := CI(k, C - TR_BACK);
  Want := Was;
  if Tr > 0 then Want := Tr;
  if (Add = 0) and (Want = Was) then exit;
  result := HeadHex(k) + HexPart(k, 13, C - TR_BACK) + HexI(Want) + HexPart(k, C - TR_BACK + 4, C) + HexI(N + Add) +
    HexPart(k, C + 4, C + 4 + N * 2) + Extra + HexPart(k, C + 4 + N * 2, CaLen[k]);
end;

// an NpcInfo (0C, without the opcode) of npc NpcId with player k's object id, position, speed, HP, his name as the
// title. The layout of this server's own NpcInfo of a monster (mask EF B3 DE FE 60): block 1 (its size, attackable,
// relations, title), block 2 (its size counting itself: id+1000000, x y z, heading, vehicle, attack / cast speed,
// move / attack speed multipliers (floats), alive, running, swim, team, 4 ints, cur HP, cur MP, max HP, max MP,
// summoned, follow (2 ints), pvp flag, reputation, state 0C = targetable + name shown)
function NpcHex(k, NpcId: integer; Atk: boolean): string;
var C, E, Hp, Mp, MHp, MMp, Run, PSp, MSp: integer;
  Mult, Spd: double;
  Title, B1, B2, Nm, Mk: string;
begin
  C := ListAt(k);
  if C >= 0 then begin
    MHp := CI(k, C - 17);
    Hp := CI(k, C - 13);
    MMp := CI(k, C - 9);
    Mp := CI(k, C - 5);
  end else begin
    MHp := 100;
    Hp := 100;
    MMp := 100;
    Mp := 100;
  end;
  E := NameEnd(k);
  Title := HexPart(k, 21, E);                         // his name, ended by two zero bytes
  Spd := 0;
  MSp := 333;
  PSp := 333;
  if E + 153 <= CaLen[k] then begin
    MSp := CI(k, E + 121);
    PSp := CI(k, E + 125);
    Run := CW(k, E + 129);
    Mult := CD(k, E + 145);
    if (Run >= 20) and (Run <= 400) and (Mult >= 0.2) and (Mult <= 5) then Spd := Run * Mult;
    if (MSp < 1) or (MSp > 5000) then MSp := 333;
    if (PSp < 1) or (PSp > 5000) then PSp := 333;
  end;
  if Spd <= 0 then Spd := 150;
  if NpcBase < 10 then NpcBase := 120;
  if LivePos(CaOID[k]) then begin
    E := PX;
    C := PY;
    Run := PZ;
  end else begin
    E := CI(k, 1);
    C := CI(k, 5);
    Run := CI(k, 9);
  end;
  if NamesOn then begin                              // his name as the npc's name, a blank title
    Nm := Title;
    Title := '20000000';                             // ' ' (empty: the client shows the npc's own "80 lvl")
    Mk := 'FFB3DEFE60';
  end else begin
    Nm := '';
    Mk := 'EFB3DEFE60';
  end;
  if Atk then B1 := '01' else B1 := '00';
  B1 := IntToHex(5 + Length(Title) div 2, 2) + B1 + '00000000' + Title;
  B2 := HexI(NpcId + 1000000) + HexI(E) + HexI(C) + HexI(Run) + HexI(0) + HexI(0) + HexI(PSp) + HexI(MSp) +
    HexF(Spd / NpcBase) + HexF(1) + '01010000' + '00000000000000000000000000000000' +
    HexI(Hp) + HexI(Mp) + HexI(MHp) + HexI(MMp) + '00' + '0000000000000000' + Nm + '00' + '00000000' + '0C';
  result := HexI(CaOID[k]) + '00' + '2500' + Mk + B1 + HexW(Length(B2) div 2 + 2) + B2;
end;

// a packet to the client as if from the server (the sniff log shows it too)
procedure ToClient(H: string);
begin
  Engine.SendToClient(H);
  if SniffOn then Lg('OUT ' + H);
end;

function InParty(OID: integer): boolean;
var j: integer;
begin
  result := false;
  for j := 0 to PtN - 1 do
    if PtOID[j] = OID then begin
      result := true;
      exit;
    end;
end;

// the party from the bot's list; true: it changed
function ReadParty(): boolean;
var i, n, OID: integer;
begin
  result := false;
  n := 0;
  try
    for i := 0 to Party.Chars.Count - 1 do
      if (i < Party.Chars.Count) and (n < 16) then begin
        OID := Party.Chars.Items(i).OID;
        if OID <> 0 then begin
          PtRead[n] := OID;
          n := n + 1;
        end;
      end;
  except
    exit;
  end;
  if n <> PtN then result := true
  else
    for i := 0 to n - 1 do
      if not InParty(PtRead[i]) then result := true;
  if not result then exit;
  for i := 0 to n - 1 do PtOID[i] := PtRead[i];
  PtN := n;
end;

// the look player k must have now -> WK (kind), WS (its signature), WA (with the target's effects)
procedure WantOf(k: integer);
begin
  WK := LK_NONE;
  WS := 0;
  WA := false;
  if not IsOn then exit;
  if InParty(CaOID[k]) then exit;                  // the party as it is
  if (MyClan <> 0) and (CaClan[k] = MyClan) and (ClanKind <> LK_NONE) then begin
    WK := ClanKind;
    WS := ClanId;
  end else if ((CaRel[k] and WAR_BITS) <> 0) and (WarKind <> LK_NONE) then begin
    WK := WarKind;
    WS := WarId;
  end else if TgOn and (CaOID[k] = CurOID) and (CurOID <> 0) then begin
    WK := LK_TR;
    WS := TgTr + 1000000;
    WA := true;
  end;
end;

// player k gets look Kind (Sig: the model / npc id) in the client
procedure Apply(k, Kind, Sig: integer; WithAve: boolean);
var P: string;
  Got, Model: integer;
  Del, Was: boolean;
begin
  CaReq[k] := Kind;
  CaSig[k] := Sig;
  Got := Kind;
  Model := Sig;
  if Model >= 1000000 then Model := Model - 1000000;
  P := '';
  if Got = LK_TR then begin
    P := TrHex(k, Model, WithAve);
    if P = '' then Got := LK_NONE;                  // nothing differs from his own (or his list not found)
  end;
  if (Got = LK_NONE) and (CaKind[k] = LK_NONE) then exit;
  Was := (CaOID[k] = CurOID) and (CurOID <> 0);
  Del := (Got = LK_NPC) or (CaKind[k] = LK_NPC);
  if Del then ToClient('08' + HexI(CaOID[k]) + '00');   // off the screen ...
  if Got = LK_NPC then ToClient('0C' + NpcHex(k, Model, ((CaRel[k] and WAR_BITS) <> 0)))
  else begin
    if Got = LK_NONE then P := OwnHex(k);
    ToClient('31' + P);
  end;
  if Del and Was then begin                          // ... the client lost the target: selected again, client only
    ToClient('B9' + '01000000' + HexI(CaOID[k]) + '0000' + '00000000');
    GuardUntil := GetTickCount + 1000;
  end;
  CaKind[k] := Got;
  if Got = LK_NPC then Lg('NPC ' + IntToStr(Model) + ' for ' + IntToStr(CaOID[k]))
  else if Got = LK_TR then Lg('MODEL ' + IntToStr(Model) + ' for ' + IntToStr(CaOID[k]))
  else Lg('OWN ' + IntToStr(CaOID[k]));
end;

// player k as he must look now (sent only when it changed)
procedure Reconcile(k: integer);
begin
  if (k < 0) or (CaOID[k] = 0) or (CaLen[k] = 0) or CaGone[k] then exit;
  WantOf(k);
  if (WK = CaReq[k]) and (WS = CaSig[k]) then exit;
  Apply(k, WK, WS, WA);
end;

// the settings changed: everyone looked at again (sent only to those whose look changed)
procedure ReconcileAll(Force: boolean);
var k: integer;
begin
  for k := 0 to CACHE_N - 1 do begin
    if Force and (CaKind[k] <> LK_NONE) then CaReq[k] := -1;
    Reconcile(k);
  end;
end;

// everyone back as the server sent him
procedure AllBack();
var k: integer;
begin
  for k := 0 to CACHE_N - 1 do
    if (CaOID[k] <> 0) and not CaGone[k] and (CaKind[k] <> LK_NONE) then Apply(k, LK_NONE, 0, false);
end;

// ===================== the sniff log =====================

function OpText(Id1, Id2: integer): string;
begin
  result := IntToHex(Id1, 2);
  if (Id1 = $FE) or (Id1 = $D0) then result := result + ':' + IntToHex(Id2, 4);
end;

procedure SniffSrv(i: integer; Note: string);
var b, n: integer;
  H: string;
begin
  if not SniffOn then exit;
  n := RgLen[i];
  H := '';
  for b := 0 to n - 1 do H := H + IntToHex(RgByte[i * RING_B + b], 2);
  if n < RgSz[i] then H := H + '..';
  Lg('S ' + OpText(RgId1[i], RgId2[i]) + ' ' + IntToStr(RgSz[i]) + Note + ' ' + H);
end;

procedure SniffCli(i: integer);
var b, n: integer;
  H: string;
begin
  n := CgLen[i];
  H := '';
  for b := 0 to n - 1 do H := H + IntToHex(CgByte[i * CRING_B + b], 2);
  if n < CgSz[i] then H := H + '..';
  Lg('C ' + OpText(CgId1[i], CgId2[i]) + ' ' + IntToStr(CgSz[i]) + ' ' + H);
end;

// ===================== the packets that came =====================

function RgInt(i, b: integer): integer;
var Base: integer;
begin
  Base := i * RING_B + b;
  result := RgByte[Base] or (RgByte[Base + 1] shl 8) or (RgByte[Base + 2] shl 16) or (RgByte[Base + 3] shl 24);
end;

// RelationChanged: a mask byte; (mask and 4) - a count (2 bytes) of records, else one; a record: OID and, when
// (mask and 1) = 0, relation (4), auto attackable (1), reputation (4), pvp flag (1)
procedure GotRelation(i: integer);
var n, Mask, Cnt, j, P, OID, V, k: integer;
begin
  n := RgLen[i];
  if n < 5 then exit;
  Mask := RgByte[i * RING_B];
  if (Mask and 1) <> 0 then exit;                    // no relation in the records
  if (Mask and 4) <> 0 then begin
    if n < 3 then exit;
    Cnt := RgByte[i * RING_B + 1] or (RgByte[i * RING_B + 2] shl 8);
    P := 3;
  end else begin
    Cnt := 1;
    P := 1;
  end;
  for j := 1 to Cnt do begin
    if P + 14 > n then exit;
    OID := RgInt(i, P);
    V := RgInt(i, P + 4);
    P := P + 14;
    if (OID = 0) or (OID = MyOID) then continue;
    RelPut(OID, V);
    k := CacheIdx(OID);
    if (k >= 0) and (CaRel[k] <> V) then begin
      CaRel[k] := V;
      Reconcile(k);
    end;
  end;
end;

procedure Drain();
var i, b, n, OID, k, k0, Base: integer;
begin
  while CgTail < CgHead do begin
    i := CgTail mod CRING_N;
    if SniffOn then SniffCli(i);
    CgTail := CgTail + 1;
  end;
  while RgTail < RgHead do begin
    i := RgTail mod RING_N;
    n := RgLen[i];
    Base := i * RING_B;
    if RgId1[i] = $08 then begin                     // DeleteObject: the player out of sight
      if n >= 4 then begin
        k := CacheIdx(RgInt(i, 0));
        if k >= 0 then begin
          CaGone[k] := true;
          CaKind[k] := LK_NONE;                      // the client has nothing of him now
          CaReq[k] := -1;
        end;
      end;
      SniffSrv(i, '');
      RgTail := RgTail + 1;
      continue;
    end;
    if RgId1[i] = $CE then begin
      SniffSrv(i, '');
      GotRelation(i);
      RgTail := RgTail + 1;
      continue;
    end;
    if RgId1[i] <> $31 then begin                    // the sniff only
      SniffSrv(i, '');
      RgTail := RgTail + 1;
      continue;
    end;
    if (n < 21) or (n < RgSz[i]) then begin          // cut (too long for the ring): not ours to touch
      SniffSrv(i, ' (CharInfo too long - left alone)');
      RgTail := RgTail + 1;
      continue;
    end;
    OID := RgInt(i, 17);
    if (OID = 0) or (OID = MyOID) then begin
      SniffSrv(i, '');
      RgTail := RgTail + 1;
      continue;
    end;
    k0 := CacheIdx(OID);
    k := CacheSlot(OID);
    if (k0 < 0) or CaGone[k] then begin              // new in sight: nothing of ours on him
      CaKind[k] := LK_NONE;
      CaClan[k] := -1;
    end;
    if k0 < 0 then CaRel[k] := RelGet(OID);
    CaOID[k] := OID;
    CaLen[k] := n;
    for b := 0 to n - 1 do CaByte[k * RING_B + b] := RgByte[Base + b];
    CaAt[k] := GetTickCount;
    CaGone[k] := false;
    if CaKind[k] = LK_TR then CaKind[k] := LK_NONE;  // the server's own replaced our copy (an npc stays: taken off first)
    CaReq[k] := -1;                                  // looked at again
    SniffSrv(i, ' (CharInfo of ' + IntToStr(OID) + ', list at ' + IntToStr(ListAt(k)) + ')');
    RgTail := RgTail + 1;
    if ListAt(k) < 0 then begin
      BadN := BadN + 1;
      if (BadN <= 5) or (BadN mod 100 = 0) then
        Lg('the list of effects not found in the CharInfo of ' + IntToStr(OID) + ' (' + IntToStr(BadN) + ' so far) hex=' +
          HexPart(k, 0, n));
    end;
    if CaClan[k] < 0 then
      if LivePos(OID) then CaClan[k] := PClan;
    Reconcile(k);
  end;
  if (RgLost + CgLost > LostSaid) and SniffOn then begin
    LostSaid := RgLost + CgLost;
    Lg('lost (the main loop was behind): ' + IntToStr(RgLost) + ' server, ' + IntToStr(CgLost) + ' client packets');
  end;
end;

// every PASS_MS: our clan (it may change), the clan of the players the bot's list did not have yet
procedure Pass();
var k, Cnt, C: integer;
begin
  C := User.ClanID;
  if C <> MyClan then begin
    MyClan := C;
    Lg('my clan ' + IntToStr(MyClan));
    ReconcileAll(false);
  end;
  if ReadParty() then begin
    Lg('party: ' + IntToStr(PtN));
    ReconcileAll(false);
  end;
  Cnt := 0;
  for k := 0 to CACHE_N - 1 do
    if (CaOID[k] <> 0) and not CaGone[k] and (CaClan[k] < 0) and (Cnt < 10) then begin
      Cnt := Cnt + 1;
      if LivePos(CaOID[k]) then begin
        CaClan[k] := PClan;
        Reconcile(k);
      end;
    end;
end;

end;

procedure SetAve(S: string);
var j, V: integer;
  W, C: string;
begin
  AveN := 0;
  W := '';
  S := S + ' ';
  for j := 1 to Length(S) do begin
    C := Copy(S, j, 1);
    if (C >= '0') and (C <= '9') then W := W + C
    else if W <> '' then begin
      V := StrToIntDef(W, 0);
      if (V > 0) and (V < 30000) and (AveN < 16) then begin
        AveIDs[AveN] := V;
        AveN := AveN + 1;
      end;
      W := '';
    end;
  end;
end;

function AveText(): string;
var j: integer;
begin
  result := '';
  for j := 0 to AveN - 1 do begin
    if result <> '' then result := result + ' ';
    result := result + IntToStr(AveIDs[j]);
  end;
  if result = '' then result := 'none';
end;

function LookText(Kind, Id: integer): string;
begin
  if Kind = LK_NPC then result := 'npc ' + IntToStr(Id)
  else if Kind = LK_TR then result := 'model ' + IntToStr(Id)
  else result := 'off';
end;

function StateText(): string;
begin
  result := 'clan: ' + LookText(ClanKind, ClanId) + '; wars: ' + LookText(WarKind, WarId) + '; target: ';
  if TgOn then result := result + 'model ' + IntToStr(TgTr) + ' + effects ' + AveText() else result := result + 'off';
  if not IsOn then result := result + ' (ALL OFF)';
end;

// '!clan npc 16019' / '!clan tr 103' / '!clan off' -> Kind, Id (false: not understood)
var NewKind, NewId: integer;
function ParseLook(S: string): boolean;
var p: integer;
  A, B: string;
begin
  result := false;
  S := Trim(S);
  p := Pos(' ', S);
  if p > 0 then begin
    A := Copy(S, 1, p - 1);
    B := Trim(Copy(S, p + 1, Length(S)));
  end else begin
    A := S;
    B := '';
  end;
  if A = 'off' then begin
    NewKind := LK_NONE;
    NewId := 0;
    result := true;
    exit;
  end;
  NewId := StrToIntDef(B, -1);
  if (NewId < 1) or (NewId >= 1000000) then exit;
  if A = 'npc' then NewKind := LK_NPC
  else if A = 'tr' then NewKind := LK_TR
  else exit;
  result := true;
end;

procedure Who();
var k, Np, Tr, Cl, Wr: integer;
begin
  Np := 0;
  Tr := 0;
  Cl := 0;
  Wr := 0;
  for k := 0 to CACHE_N - 1 do
    if (CaOID[k] <> 0) and not CaGone[k] then begin
      if CaKind[k] = LK_NPC then Np := Np + 1;
      if CaKind[k] = LK_TR then Tr := Tr + 1;
      if (MyClan <> 0) and (CaClan[k] = MyClan) then Cl := Cl + 1;
      if (CaRel[k] and WAR_BITS) <> 0 then Wr := Wr + 1;
    end;
  Say('in sight (seen since the start): clan ' + IntToStr(Cl) + ', wars ' + IntToStr(Wr) + '; shown as npc ' +
    IntToStr(Np) + ', as a model ' + IntToStr(Tr) + '. My clan id ' + IntToStr(MyClan));
end;

procedure Command(C: string);
var W, S: string;
  p, V: integer;
begin
  p := Pos(' ', C);
  if p > 0 then begin
    W := Copy(C, 1, p - 1);
    S := Trim(Copy(C, p + 1, Length(C)));
  end else begin
    W := C;
    S := '';
  end;
  if (W = 'clan') or (W = 'war') then begin
    if S = '' then begin
      Say(StateText());
      exit;
    end;
    if not ParseLook(S) then begin
      Say('!' + W + ' npc 16019 | !' + W + ' tr 103 | !' + W + ' off');
      exit;
    end;
    if W = 'clan' then begin
      ClanKind := NewKind;
      ClanId := NewId;
    end else begin
      WarKind := NewKind;
      WarId := NewId;
    end;
    ReconcileAll(true);
    Say(StateText());
  end else if (W = 'big') or (W = 'head') or (W = 'both') or (W = 'ave') then begin
    if W = 'big' then SetAve('17')
    else if W = 'head' then SetAve('14')
    else if W = 'both' then SetAve('17 14')
    else if S = '0' then AveN := 0
    else begin
      SetAve(S);
      if AveN = 0 then begin
        Say('!ave needs numbers: !ave 17 14 (!ave 0 - none)');
        exit;
      end;
    end;
    TgOn := true;
    ReconcileAll(true);
    Say(StateText());
  end else if W = 'tr' then begin
    if S = '' then begin
      Say('models: 104 pig, 105 rabbit, 111 frog, 114 snowman, 108 pumpkin, 112 child, 103 buffalo, 305 zaken, ' +
        '302 akamanah, 301 zariche, 311 demon prince, 143 queen ant, 216 dragon bomber, 1 onyx beast; !tr 0 - own');
      exit;
    end;
    V := StrToIntDef(S, -1);
    if (V < 0) or (V > 30000) then begin
      Say('!tr needs a number: !tr 104');
      exit;
    end;
    TgTr := V;
    TgOn := true;
    ReconcileAll(true);
    Say(StateText());
  end else if W = 'target' then begin
    TgOn := not TgOn;
    ReconcileAll(false);
    Say(StateText());
  end else if W = 'speed' then begin
    V := StrToIntDef(S, -1);
    if (V < 10) or (V > 1000) then begin
      Say('!speed N (10..1000, now ' + IntToStr(NpcBase) + '): an npc runs behind the player - less, ahead - more');
      exit;
    end;
    NpcBase := V;
    ReconcileAll(true);
    Say('npc base speed ' + IntToStr(NpcBase) + ' - the npcs made anew');
  end else if W = 'names' then begin
    NamesOn := not NamesOn;
    ReconcileAll(true);
    if NamesOn then Say('npcs with the players'' names (/target Name works)')
    else Say('npcs with their own names, the player''s name as the title');
  end else if W = 'who' then Who()
  else if W = 'sniff' then begin
    SniffOn := not SniffOn;
    if SniffOn then Say('sniff on - every packet into the log') else Say('sniff off');
  end else if W = 'off' then begin
    AllBack();
    IsOn := false;
    ReconcileAll(false);
    Say('off - everyone as the server sent him');
  end else if W = 'on' then begin
    IsOn := true;
    ReconcileAll(true);
    Say(StateText());
  end else if W = 'dump' then DumpTarget()
  else Say('commands: !clan npc N|tr N|off, !war npc N|tr N|off, !target, !tr N, !big !head !ave N, !speed N, ' +
    '!names, !who, !off !on, !sniff, !dump');
end;
