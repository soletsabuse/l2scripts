// PvP focus: keys that target enemy players. Everything of it lives in its own folder Library\PvP:
//   pvp_focus.txt      this script (run it in the bot on every character)
//   pvp.html           the page - opened through the PvP server: http://127.0.0.1:7790
//   pvp_server.py      the PvP server (pvp_server.bat starts it with no window and opens the page)
//   profiles\<id>.pvp  the profiles, any number of them
//   pvp.ini            active=<id>: the profile of every character without one of his own; a profile named like
//                      the character (profiles\<his name>.pvp, any case) is his own
// The script reads the profile files itself (a change counts within a second, no restart) - the keys work with
// the server down too. With the server up it is told at once (RELOAD) when the page saves, and it sends what it
// sees (STATE: the profile it uses and what it made of it, the players around - clan, class, place, cast, revive,
// the skills cast nearby, our own skills) ~3 times a second: the page is live.
//
// A group of a profile: a key (none: auto only) and a mode:
//   class   - enemies of the classes of the group, by the class priority (1 first), then the nearest
//   reskill - enemies up from the dead res_min_ms..res_max_ms ago (the classes: optional). Its key turns it ON / OFF
//             (auto=1: on at the start); while on, the bot takes such an enemy by itself (right after each look at who
//             got up, every REV_SCAN_MS) and does the after action at once (attack). Only your key press a moment ago
//             holds it (not our cast, not the after skill / volley being sent); then it waits, at most RES_LATE_MS.
//             The one it took: assist / volley leave him to us for RES_KEEP_MS while he lives and is our target. kill_sound=<file.wav> (in Library\PvP\sounds): a successful res-kill
//             - an enemy up from the dead dies again within kill_ms (default 20000) with no buffs seen on him (fewer
//             than KILL_BUFFED; the server may not show enemy buffs at all) -> that sound + a notice
//   onus    - enemies holding me or a party member in their target (the ones on me first; the classes: optional)
//   assist  - the target of a player: assist=<1st>|<2nd>|<3rd> ('' = the party leader) - the first of them in sight, alive
//             and (assist_aggro 2) not aggroed; aggroed / killed / gone -> the next one. Its key turns it ON / OFF
//             (auto=1: on at the start); while on, it follows: his target now, then his target changed (or the next
//             one took over) -> taken at once + the after action. The list is a chain: a character on it follows
//             only the ones above him (the first one leads himself, the next one leads when the ones above are out).
//             volley=<ID>|..: the one we follow starts one of these skills -> the same skill of ours at his target at
//             once (a volley; only when ours is ready; ctrl= of the group)
//             assist_aggro: 2 an aggroed assister is skipped (an enemy in sight holds him in the target), 1 no assist
//             while I am aggroed, 0 no check
//             assist_src=net (assist via the server): ONE assister (the first name; none: the party leader), any
//             character whose bot is on the relay (in the party or not). Not his game target - his bot sends it
//             (PvpPanel -> the relay -> the PvpPanel of every PC; the same PC: straight from the panel), so he may be
//             far, out of sight, dead. His target is taken as it is: no range / height, no aggro, no holds (only not
//             before the game knows it: in sight); at a friend: selected, no after action. Every bot sends its own
//             target as soon as it changes (N|T), whatever its groups
// CP potions (cp_item, '0' none): a thread of its own looks at our CP every CP_LOOK_MS; below cp_pct % -> the potion,
// again not sooner than cp_gap_ms (the server's flood guard); the potions left: looked at every second (told when none)
//   cast    - enemies casting one of the skills (Battle Force 426, Spell Force 427, a resurrection ...):
//             who=target - the one it is cast on (the one being charged: the most casters first),
//             who=caster - the caster (the classes: optional, of the one taken)
//   names   - the priority players of the list, in its order (any class / clan)
//   smart   - the assisters: the enemy who switches his target onto one of us (me / my party) first while the
//             others switch onto the same one after him (an assist call). Only the target switches count - a cast,
//             an AoE, a hit do not; switches at the same moment as his (< ASSIST_MIN_MS: an AoE, an auto target)
//             do not either. Points: one per follower (1.5 when the one switched onto is me), halved every
//             ASSIST_HALF_MS. The classes: optional. None with points in range -> the one targeting us now
// Every mode: the priority players of the list first (prio_first=1; default 0), the skipped ones never; never a party
// member / our clan / our alliance; within the group's range (range=0: the profile's range= of the general part,
// default 1200) and height. An enemy: his clan OR alliance is
// on the clans list (any case; the list empty -> anyone), or he is a priority player.
// Pressed again while on its last pick -> the next one by the same order (a cycle: it ends cycle_ms after the
// last press or when the target is changed by other means). Only the target now can be taken -> its after
// action again.
// auto=1 (cast): the group fires by itself on a new cast of its skills (not more often than auto_ms, not while the
// target is one of the group already). reskill: see above.
// The auto picks (cast auto, assist) never cut in on you: not within AUTO_HOLD_MS of a key pick, not while the after
// skill of a pick is still being sent, not while our own cast goes; they do not touch the cycle of a key. Res-kill
// waits only for a key pick (its window is short). A res-kill
// turned on / off by its key stays so when the profile is read again (the page saved).
// The notices (what was taken, res-kill on / off): notify=<chat type> - a fake chat message only we see: 15 the command
// channel commander (in the middle of the screen, as the siege commander's), 19 a screen announce, 3 party, 0 none.
// A pick: the target; mark=1..4 -> that token on him (RequestActionUse $4E..$51, sniffed with token_sniffer);
// after=attack -> the auto attack, after=skill:<ID> -> that skill (sent every RESEND_MS till it starts, up to
// FIRE_MS); ctrl=1 -> both as with Ctrl held (default 0). Said in the bot window and (chat=1) as a fake party chat
// message only we see. A key works only while the game window of THIS character is the active one.
// The casts are read as force_sounds does (CharList(i).Cast / .Target every CAST_SCAN_MS); the revives:
// a player seen dead, then alive within REV_GAP_MS (in sight all the time - not one back from the town later).
//
// The keys come first: the scans for the page (the state, the casts, the revives) look at the keys every
// SCAN_CHUNK players and stop for a press (done the next time). The state is built only while a page watches
// (the server says WATCH|1 / WATCH|0; an old server says nothing - always). The names / clans of the players
// are read from the bot once per NAME_KEEP_MS (its string fields read often broke scripts), not every state.
// "perf" in the state: the press -> SetTarget time, the slowest state / cast scan / loop of the last PERF_KEEP_MS.
//
// The stats (the page's Stats tab; "foes" in the state every FOES_MS): every player in sight followed from the
// bot's own data (CharList(i).Target, .Cast - no packets, no system messages): how often an enemy switched onto

// =============================================================================

procedure SplitTo(S, Sep: string; L: TStringList);
var p: integer;
begin
  L.Clear;
  if S = '' then exit;
  while true do begin
    p := Pos(Sep, S);
    if p = 0 then begin L.Add(S); break; end;
    L.Add(Copy(S, 1, p - 1));
    Delete(S, 1, p);
  end;
end;

function JsonStr(S: string): string;
var i: integer;
  C: string;
begin
  result := '"';
  for i := 1 to Length(S) do begin
    C := Copy(S, i, 1);
    if (C = '"') or (C = '\') then result := result + '\' + C
    else if Ord(S[i]) < 32 then result := result + ' '
    else result := result + C;
  end;
  result := result + '"';
end;

function HexLE32(V: integer): string;
begin
  result := IntToHex(V and $FF, 2) + IntToHex((V shr 8) and $FF, 2) +
            IntToHex((V shr 16) and $FF, 2) + IntToHex((V shr 24) and $FF, 2);
end;

// the character name / a profile id -> a file name (letters, digits, _ and - only)
function FileTag(S: string): string;
var i: integer;
  C: string;
begin
  result := '';
  for i := 1 to Length(S) do begin
    C := Copy(S, i, 1);
    if ((C >= 'a') and (C <= 'z')) or ((C >= 'A') and (C <= 'Z')) or ((C >= '0') and (C <= '9')) or
       (C = '_') or (C = '-') then result := result + C
    else result := result + '_';
  end;
  if result = '' then result := 'char';
end;

function KeyVK(S: string): integer;
var n: integer;
begin
  result := 0;
  S := UpperCase(Trim(S));
  if S = '' then exit;
  if (Length(S) = 1) and (((S[1] >= 'A') and (S[1] <= 'Z')) or ((S[1] >= '0') and (S[1] <= '9'))) then begin
    result := Ord(S[1]);
    exit;
  end;
  if (Length(S) >= 2) and (S[1] = 'F') then begin
    n := StrToIntDef(Copy(S, 2, 2), 0);
    if (n >= 1) and (n <= 24) then begin result := $6F + n; exit; end;
  end;
  if Copy(S, 1, 3) = 'NUM' then begin
    n := StrToIntDef(Copy(S, 4, 1), -1);
    if (n >= 0) and (Length(S) = 4) then begin result := $60 + n; exit; end;
  end;
  if S = 'SPACE' then result := $20
  else if S = 'TAB' then result := $09
  else if S = 'ENTER' then result := $0D
  else if S = 'ESC' then result := $1B
  else if S = 'BACKSPACE' then result := $08
  else if S = 'INSERT' then result := $2D
  else if S = 'DELETE' then result := $2E
  else if S = 'HOME' then result := $24
  else if S = 'END' then result := $23
  else if S = 'PAGEUP' then result := $21
  else if S = 'PAGEDOWN' then result := $22
  else if S = 'UP' then result := $26
  else if S = 'DOWN' then result := $28
  else if S = 'LEFT' then result := $25
  else if S = 'RIGHT' then result := $27
  else if S = 'MOUSE3' then result := $04
  else if S = 'MOUSE4' then result := $05
  else if S = 'MOUSE5' then result := $06
  else if S = '~' then result := $C0
  else if S = 'MINUS' then result := $BD
  else if S = 'EQUAL' then result := $BB
  else if S = 'LBRACKET' then result := $DB
  else if S = 'RBRACKET' then result := $DD
  else if S = 'SEMICOLON' then result := $BA
  else if S = 'QUOTE' then result := $DE
  else if S = 'COMMA' then result := $BC
  else if S = 'PERIOD' then result := $BE
  else if S = 'SLASH' then result := $BF
  else if S = 'BACKSLASH' then result := $DC
  else if S = 'NUMMUL' then result := $6A
  else if S = 'NUMADD' then result := $6B
  else if S = 'NUMSUB' then result := $6D
  else if S = 'NUMDOT' then result := $6E
  else if S = 'NUMDIV' then result := $6F;
end;

// "Ctrl+Shift+Z" -> VK + 256 * (1 Ctrl, 2 Shift, 4 Alt); 0 = cannot read it
function ParseKey(S: string): integer;
var L: TStringList;
  i, Md, VK: integer;
  P: string;
begin
  result := 0;
  Md := 0;
  VK := 0;
  L := TStringList.Create;
  SplitTo(Trim(S), '+', L);
  for i := 0 to L.Count - 1 do begin
    P := UpperCase(Trim(L[i]));
    if P = 'CTRL' then Md := Md or 1
    else if P = 'SHIFT' then Md := Md or 2
    else if P = 'ALT' then Md := Md or 4
    else VK := KeyVK(P);
  end;
  L.Free;
  if VK <> 0 then result := VK + 256 * Md;
end;

function KeyDown(VK: integer): boolean;
begin
  result := (GetAsyncKeyState(VK) and $8000) <> 0;
end;

// modifier m: 0 Ctrl, 1 Shift, 2 Alt
function ModVK(m: integer): integer;
begin
  if m = 0 then result := $11
  else if m = 1 then result := $10
  else result := $12;
end;

// the modifiers of a group key (Md: 1 Ctrl, 2 Shift, 4 Alt): the needed ones down (Grace: or seen down
// within MOD_GRACE_MS - let go a moment before the key), the others up
function ModsOk(Md: integer; Grace: boolean): boolean;
var m: integer;
  DownNow, Recent: boolean;
begin
  result := false;
  for m := 0 to 2 do begin
    DownNow := KeyDown(ModVK(m));
    if (Md and (1 shl m)) <> 0 then begin
      Recent := Grace and (ModAt[m] <> 0) and (GetTickCount - ModAt[m] <= MOD_GRACE_MS);
      if not (DownNow or Recent) then exit;
    end else if DownNow then exit;
  end;
  result := true;
end;

// a group key pressed (Src: 1 a game key event, 2 a look). The same press caught by both ways is one;
// two presses caught the same way are two (a quick double press cycles twice). Numbers only (both threads)
procedure Trigger(g, Src: integer);
begin
  if (g = LastG) and (Src <> LastSrc) and (GetTickCount - LastAt < SAME_MS) then exit;
  LastG := g;
  LastSrc := Src;
  LastAt := GetTickCount;
  NextG := g;
end;

// a look at the keys (main loop): a group key down now and not the last time, or tapped between two looks
procedure PollKeys();
var g, m, VK, St: integer;
  Focused, DownNow, Tapped: boolean;
begin
  for m := 0 to 2 do
    if KeyDown(ModVK(m)) then ModAt[m] := GetTickCount;
  Focused := GetForegroundWindow() = Engine.GameWindow;   // only in the game window of this character
  for g := 0 to GroupN - 1 do begin
    if GKey[g] = 0 then continue;   // auto only
    VK := GKey[g] mod 256;
    St := GetAsyncKeyState(VK);
    DownNow := (St and $8000) <> 0;
    Tapped := (St and 1) <> 0;
    if Focused and (not GWas[g]) and (DownNow or Tapped) and ModsOk(GKey[g] div 256, true) then
      Trigger(g, 2);
    GWas[g] := DownNow;
  end;
end;

// the key events of the game window (the bot's laKey, p1 = the key): a group key -> its group, if that
// key is really down / was just tapped (an event once came with the code of another key). A held key
// repeats: a new press only after REPEAT_MS without its events. Numbers only here (no strings: two
// threads building strings at once crash PaxScript)
procedure KeyEvents_Thread();
var KeyCode, P2, g, Code, St: integer;
begin
  while true do begin
    KeyCode := 0;
    if Engine.WaitAction([laKey], KeyCode, P2, 1000) <> laKey then continue;
    if GetForegroundWindow() <> Engine.GameWindow then continue;
    for g := 0 to GroupN - 1 do begin
      Code := GKey[g];
      if Code = 0 then continue;
      if KeyCode <> Code mod 256 then continue;
      St := GetAsyncKeyState(Code mod 256);
      if (St and $8001) = 0 then continue;   // not down, not tapped: not a press of this key
      if GetTickCount - EvAt[g] > REPEAT_MS then
        if ModsOk(Code div 256, false) then Trigger(g, 1);
      EvAt[g] := GetTickCount;
    end;
  end;
end;

// a press waiting (the scans stop for it): the keys looked at now
function KeyPending(): boolean;
begin
  if NextG < 0 then PollKeys();
  result := NextG >= 0;
end;

// somebody got up from the dead res_min_ms..res_max_ms of group g ago (before the whole scan: cheap)

end;

// who is dead now, who got up since the last look (numbers only). Got up = alive now and seen dead within
// REV_GAP_MS: one seen dead long ago (he went to the town, came back) is not a revive. Stops for a key press
procedure ScanRevives();
var i, k, OID: integer;
begin
  for i := 0 to CharList.Count - 1 do if i < CharList.Count then begin
    if (i mod SCAN_CHUNK = SCAN_CHUNK - 1) and KeyPending() then exit;
    OID := CharList(i).OID;
    if OID = 0 then continue;
    if CharList(i).Dead then RememberDead(OID)
    else begin
      k := DeadIdx(OID);
      if k >= 0 then begin
        if GetTickCount - DeadAt[k] <= REV_GAP_MS then RememberRevive(OID);
        DeadOID[k] := 0;
        DeadCnt := DeadCnt - 1;
      end;
    end;
  end;
end;

// '"n": .., "c": .., "a": ..' of player i (OID) for the state: read from the bot once per NAME_KEEP_MS. The place:
// OID mod NAME_SLOTS, the next ones (up to 8) when taken by another; all taken -> the first one is reused
function NameJson(i, OID: integer): string;
var h, j, s, Spot: integer;
begin
  h := OID mod NAME_SLOTS;

  Spot := -1;
  for j := 0 to 7 do begin

      if GetTickCount - NcAt[s] < NAME_KEEP_MS then begin
        result := NcJs[s];
        exit;
      end;
      Spot := s;
      break;
    end;
    if (Spot < 0) and ((NcOID[s] = 0) or (GetTickCount - NcAt[s] > 60000)) then Spot := s;
  end;
  if Spot < 0 then Spot := h;
  result := '"n": ' + JsonStr(CharList(i).Name) + ', "c": ' + JsonStr(CharList(i).Clan) + ', "a": ' +
    JsonStr(CharList(i).Ally);
  NcOID[Spot] := OID;
  NcJs[Spot] := result;
  NcAt[Spot] := GetTickCount;
end;

// a slot for player i (OID, new): a free place of his chain, else the one seen the longest ago there
function FoSlot(i, OID: integer): integer;
var h, j, s, Old: integer;
  OldAt: cardinal;
begin
  h := OID mod FOE_SLOTS;
  if h < 0 then h := -h;
  Old := h;
  OldAt := $FFFFFFFF;
  result := -1;
  for j := 0 to 15 do begin
    s := (h + j) mod FOE_SLOTS;
    if FoOID[s] = 0 then begin
      result := s;
      break;
    end;
    if FoSeen[s] < OldAt then begin
      OldAt := FoSeen[s];
      Old := s;
    end;
  end;
  if result < 0 then result := Old;
  FoOID[result] := OID;
  FoClear(result);
  FoCls[result] := CharList(i).ClassID;
  FoOurs[result] := IsOurs(i);
  FoOursAt[result] := GetTickCount;
  FoJs[result] := '';
end;

// 2 ^ (-dt / ASSIST_HALF_MS): the whole halvings, the rest straight (close enough)
function HalfOf(dt: cardinal): double;
var n, j: integer;
begin
  result := 1;
  n := dt div ASSIST_HALF_MS;
  if n > 20 then begin
    result := 0;
    exit;
  end;
  for j := 1 to n do result := result * 0.5;
  result := result * (1 - ((dt mod ASSIST_HALF_MS) / ASSIST_HALF_MS) / 2);
end;

// the assister points of slot k now
function ScoreOf(k: integer): double;
begin
  result := FoScore[k] * HalfOf(GetTickCount - FoScAt[k]);
end;

procedure AddScore(k: integer; W: double);
begin
  FoScore[k] := ScoreOf(k) + W;
  FoScAt[k] := GetTickCount;
end;

// the burst on V (one of us): its place (made: the oldest place reused)
function BurstOf(V: integer): integer;
var j, Old: integer;
begin
  Old := 0;
  for j := 0 to 15 do begin
    if BuV[j] = V then begin
      result := j;
      exit;
    end;
    if BuAt[j] < BuAt[Old] then Old := j;
  end;
  result := Old;
  BuV[result] := V;
  BuInit[result] := 0;
  BuN[result] := 0;
  BuMemN[result] := 0;
  BuAt[result] := 0;
end;

function InBurst(j, OID: integer): boolean;
var m: integer;
begin
  result := false;
  for m := 0 to BuMemN[j] - 1 do
    if BuMem[j * 16 + m] = OID then begin
      result := true;
      exit;
    end;
end;

// the enemy of slot k switched onto V (one of us) now: the first of a new burst on V, or one more of it - a
// follower of the first when later than ASSIST_MIN_MS and the first still holds V (an assist takes his target)
procedure SwitchToUs(k, V: integer);
var j, f, OID: integer;
  W: double;
begin
  OID := FoOID[k];
  j := BurstOf(V);
  if (BuInit[j] = 0) or (GetTickCount - BuAt[j] > ASSIST_WIN_MS) then begin
    BuInit[j] := OID;
    BuAt[j] := GetTickCount;
    BuN[j] := 0;
    BuMem[j * 16] := OID;
    BuMemN[j] := 1;
    exit;
  end;
  if InBurst(j, OID) then exit;                    // back onto him: counted already
  if BuMemN[j] < 16 then begin
    BuMem[j * 16 + BuMemN[j]] := OID;
    BuMemN[j] := BuMemN[j] + 1;
  end;
  if GetTickCount - BuAt[j] < ASSIST_MIN_MS then exit;   // the same moment as the first: an AoE / an auto target
  f := FoIdx(BuInit[j]);
  if f < 0 then exit;
  if FoTgt[f] <> V then exit;                      // the first is on another one now: not his assist
  BuN[j] := BuN[j] + 1;
  if BuN[j] = 1 then FoInit[f] := FoInit[f] + 1;
  FoLed[f] := FoLed[f] + 1;
  W := 1;
  if V = MyOID then W := 1.5;                      // onto me: more
  AddScore(f, W);
  FoLast[f] := GetTickCount;
  FoFollow[k] := FoFollow[k] + 1;
end;

// player i (OID) targets Tg now. A switch of an enemy onto us: counted, the assist bursts. A player new / back in
// sight: his target taken as it is (no switch)
procedure TrackTarget(i, OID, Tg: integer);
var k: integer;
begin
  k := FoIdx(OID);
  if k < 0 then begin
    k := FoSlot(i, OID);
    FoSeen[k] := 0;
  end;
  if FoJs[k] = '' then FoJs[k] := NameJson(i, OID);
  if GetTickCount - FoOursAt[k] > 10000 then begin  // the party / the clan may change
    FoOurs[k] := IsOurs(i);
    FoOursAt[k] := GetTickCount;
  end;
  if GetTickCount - FoSeen[k] > FOE_SEEN_MS then begin   // new / back in sight
    FoSeen[k] := GetTickCount;
    FoTgt[k] := Tg;
    FoTgtAt[k] := GetTickCount;
    exit;
  end;
  FoSeen[k] := GetTickCount;
  if Tg = FoTgt[k] then exit;
  if FoTgt[k] = MyOID then FoTMe[k] := FoTMe[k] + integer(GetTickCount - FoTgtAt[k]);
  FoTgt[k] := Tg;
  FoTgtAt[k] := GetTickCount;
  if FoOurs[k] or (Tg = 0) then exit;
  if Tg = MyOID then FoSwMe[k] := FoSwMe[k] + 1
  else if IsUs(Tg) then FoSwUs[k] := FoSwUs[k] + 1
  else exit;
  FoLast[k] := GetTickCount;
  SwitchToUs(k, Tg);
end;

// a new cast of OID (skill Sk) at Tg: at one of us -> counted, its skill tallied (the 4 most cast kept)
procedure CastAtUs(OID, Sk, Tg: integer);
var k, j, Least: integer;
begin
  if not IsUs(Tg) then exit;
  k := FoIdx(OID);
  if k < 0 then exit;
  if FoOurs[k] then exit;
  if Tg = MyOID then FoCastMe[k] := FoCastMe[k] + 1 else FoCastUs[k] := FoCastUs[k] + 1;
  FoLast[k] := GetTickCount;
  Least := 0;
  for j := 0 to 3 do begin
    if FoSk[k * 4 + j] = Sk then begin
      FoSkN[k * 4 + j] := FoSkN[k * 4 + j] + 1;
      exit;
    end;
    if FoSkN[k * 4 + j] < FoSkN[k * 4 + Least] then Least := j;
  end;
  FoSk[k * 4 + Least] := Sk;
  FoSkN[k * 4 + Least] := 1;
end;

  end;

// who casts what now (as force_sounds reads it): a cast seen within CAST_KEEP_MS still counts. A new one (a
// caster not casting that skill before) of a skill of an auto cast group -> that group fires
procedure CastScan(Abortable: boolean);
var i, k, g, n, OID, Sk, Tg, Cls: integer;
  IsNew: boolean;
  T: TL2Live;
begin
  n := 0;
  for k := 0 to CsN - 1 do
    if GetTickCount - CsAt[k] <= CAST_KEEP_MS then begin
      if n <> k then begin
        CsOID[n] := CsOID[k];
        CsSk[n] := CsSk[k];
        CsTgt[n] := CsTgt[k];
        CsCls[n] := CsCls[k];
        CsAt[n] := CsAt[k];
      end;
      n := n + 1;
    end;
  CsN := n;
  for i := 0 to CharList.Count - 1 do if i < CharList.Count then begin
    if Abortable and (i mod SCAN_CHUNK = SCAN_CHUNK - 1) and KeyPending() then exit;   // the key first
    if CharList(i).Dead then continue;
    OID := CharList(i).OID;
    if (OID = 0) or (OID = MyOID) then continue;
    Tg := 0;
    T := CharList(i).Target;                        // the object once (it may go meanwhile)
    if T <> nil then Tg := T.OID;
    TrackTarget(i, OID, Tg);                        // who switched onto us
    Sk := 0;
    try
      if CharList(i).Cast.EndTime > 0 then Sk := CharList(i).Cast.ID;
    except
      Sk := 0;
    end;
    if Sk <= 0 then continue;
    Cls := CharList(i).ClassID;
    k := CastIdx(OID);
    IsNew := true;
    if k >= 0 then IsNew := CsSk[k] <> Sk;
    if k < 0 then begin
      if CsN >= MAX_CS then continue;
      k := CsN;
      CsN := CsN + 1;
    end;
    CsOID[k] := OID;
    CsSk[k] := Sk;
    CsTgt[k] := Tg;
    CsCls[k] := Cls;
    CsAt[k] := GetTickCount;
    if not IsNew then continue;
    CastAtUs(OID, Sk, Tg);
    if not IsOurs(i) then NoteSeen(Sk, Cls);
    for g := 0 to GroupN - 1 do
      if GAuto[g] and (GMode[g] = MODE_CAST) then
        if SkInGroup(g, Sk) then AutoWant[g] := true;
    // a volley skill: an assist group that is on has it -> VolleyTick (is he the one it follows?)
    for g := 0 to GroupN - 1 do
      if (VolG < 0) and (GMode[g] = MODE_ASSIST) and GResOn[g] then
        if VolHas(g, Sk) then begin
          VolG := g;
          VolOID := OID;
          VolSk := Sk;
          VolTgt := Tg;
        end;
  end;
end;

function InCycle(OID: integer): boolean;
var k: integer;
begin
  result := false;
  for k := 0 to CycN - 1 do
    if CycOID[k] = OID then begin
      result := true;
      exit;
    end;
end;

// the group g once over the players. UseCycle: the ones picked in this cycle are skipped; the target now
// (Cur) is never the pick. Gives the best one left (-1 none; BestOID, BestD). The order: the priority
// player first (by the list), then the class priority, then (cast, who=target) the most casters on him,
// then the nearest. InRange: how many can be taken (Cur too), CurIn: Cur is one of them; FarI / FarD: the
// nearest one of the group out of range / height (-1 none)
function ScanGroup(g, Cur: integer; UseCycle: boolean; var BestOID: integer; var BestD: double;
  var InRange: integer; var CurIn: boolean; var FarI: integer; var FarD: double): integer;
var i, OID, Cls, PX, PY, PZ, T, Rk, Tag, Cnt, Md, k, Tg: integer;
  D, Sc, BestSc, Sm: double;
  OnUs: boolean;
  TL: TL2Live;
  Ago: cardinal;
begin
  result := -1;
  BestOID := 0;
  BestD := 0;
  BestSc := 1000000000000.0;
  InRange := 0;
  CurIn := false;
  FarI := -1;
  FarD := 1000000;
  Md := GMode[g];
  for i := 0 to CharList.Count - 1 do if i < CharList.Count then begin
    if CharList(i).Dead then continue;
    OID := CharList(i).OID;
    if (OID = 0) or (OID = MyOID) then continue;
    Cls := CharList(i).ClassID;
    T := 1;
    Cnt := 0;
    if (Md <> MODE_NAMES) and ((Md = MODE_CLASS) or (GClsN[g] > 0)) then begin
      T := TierOf(g, Cls);
      if T = 0 then continue;
    end;
    Sm := 0;
    if Md = MODE_RES then begin
      Ago := RevivedAgo(OID);
      if (Ago < cardinal(GResMin[g])) or (Ago > cardinal(GResMax[g]) + ResGrace) then continue;
    end else if Md = MODE_ONUS then begin
      TL := CharList(i).Target;                      // whom he holds in his target: me / a party member
      Tg := 0;
      if TL <> nil then Tg := TL.OID;
      if not IsUs(Tg) then continue;
      if Tg <> MyOID then Cnt := 1;                  // on a party member: after the ones on me
    end else if Md = MODE_SMART then begin
      k := FoIdx(OID);
      OnUs := false;
      if k >= 0 then begin
        Sm := ScoreOf(k);
        OnUs := IsUs(FoTgt[k]);
      end;
      if SmartFb then begin
        if not OnUs then continue;                   // the fallback: one who targets us now
      end else if Sm < SMART_MIN then continue;
    end else if Md = MODE_CAST then begin
      if GWho[g] = 1 then begin
        if not CastingFor(g, OID) then continue;
      end else begin
        Cnt := ChargeCount(g, OID);
        if Cnt = 0 then continue;
      end;
    end;
    if IsOurs(i) then continue;
    Rk := 99;
    Tag := 0;
    if PlN > 0 then Tag := PlayerTag(LowerCase(Trim(CharList(i).Name)), Rk);
    if Tag = 2 then continue;
    if Md = MODE_NAMES then begin
      if Tag <> 1 then continue;
    end else begin
      if Tag <> 1 then
        if not IsFoeClan(i) then continue;
      if not GPrio[g] then Rk := 0;
    end;
    PX := CharList(i).X;
    PY := CharList(i).Y;
    PZ := CharList(i).Z;
    D := User.DistTo(PX, PY, PZ);
    if (D > GRange[g]) or (Abs(PZ - User.Z) > GHeight[g]) then begin
      if D < FarD then begin
        FarD := D;
        FarI := i;
      end;
      continue;
    end;
    InRange := InRange + 1;
    if OID = Cur then begin
      CurIn := true;
      continue;
    end;
    if UseCycle then
      if InCycle(OID) then continue;
    if Cnt > 20 then Cnt := 20;
    if Sm > 99 then Sm := 99;
    if Md = MODE_SMART then Sc := Rk * 100000000.0 + T * 10000000.0 + (99 - Sm) * 100000.0 + D
    else if Md = MODE_ONUS then Sc := Rk * 100000000.0 + T * 10000000.0 + Cnt * 1000000.0 + D
    else Sc := Rk * 100000000.0 + T * 10000000.0 + (20 - Cnt) * 100000.0 + D;
    if Sc < BestSc then begin
      BestSc := Sc;
      BestD := D;
      BestOID := OID;
      result := i;
    end;
  end;
end;

// player i (if he is still OID) as "Name [Clan]"
function WhoText(i, OID: integer): string;
var C: string;
begin
  result := '?';
  if (i < 0) or (i >= CharList.Count) then exit;
  if CharList(i).OID <> OID then exit;
  result := CharList(i).Name;
  C := CharList(i).Clan;
  if C <> '' then result := result + ' [' + C + ']';
end;

// token T (1..4) on the target: RequestActionUse $4E..$51 (sniffed with token_sniffer)
procedure SendMark(T: integer);
begin
  if (T < 1) or (T > 4) then exit;
  Engine.SendToServer('56' + HexLE32($4D + T) + '00000000' + '00');
end;

// ms of reuse left of our skill ID (0 = ready / not ours)
function SkLeft(ID: integer): integer;
var Sk: TL2Skill;
begin
  result := 0;
  if SkillList.ByID(ID, Sk) then result := Sk.EndTime;
end;

// our cast of skill ID: ms left of it (0 = not casting it)
function CastLeft(ID: integer): integer;
var CID, Left: integer;
begin
  result := 0;
  Left := User.Cast.EndTime;
  if Left <= 0 then exit;
  CID := User.Cast.ID;
  if CID = ID then result := Left;
end;

// the after skill being sent: started (its reuse grew / our cast of it) -> done; the time is up / the target
// switched (not right at the start: the target may not be ours yet) / we are dead -> given up
procedure AfterTick();
begin
  if PendSk = 0 then exit;
  if (SkLeft(PendSk) > PendL0) or (CastLeft(PendSk) > 0) then begin
    PendSk := 0;
    exit;
  end;
  if (GetTickCount - PendStart > FIRE_MS) or User.Dead then begin
    Engine.Msg('[PvP]', 'skill ' + IntToStr(PendSk) + ' did not start', 0);
    PendSk := 0;
    exit;
  end;
  if GetTickCount - PendStart > 400 then
    if TargetOID() <> PendTgt then begin
      PendSk := 0;
      exit;
    end;
  if GetTickCount - PendSent >= RESEND_MS then begin
    Engine.DUseSkill(PendSk, PendCtrl, false);
    PendSent := GetTickCount;
  end;
end;

// the after action of group g at OID (just targeted): ' -> what' for the message
function DoAfter(g, OID: integer): string;
var Sk: TL2Skill;
  Left: integer;
begin
  result := '';
  if GAfter[g] = 1 then begin
    Engine.Attack(300, GCtrl[g]);
    result := ' -> атака';
    exit;
  end;
  if (GAfter[g] <> 2) or (GAfterSk[g] <= 0) then exit;
  if not SkillList.ByID(GAfterSk[g], Sk) then begin
    result := ' -> нет скилла ' + IntToStr(GAfterSk[g]);
    exit;
  end;
  Left := Sk.EndTime;
  if Left > 1000 then begin
    result := ' -> ' + Sk.Name + ' в откате ' + IntToStr((Left + 999) div 1000) + ' с';
    exit;
  end;
  PendSk := GAfterSk[g];
  PendTgt := OID;
  PendL0 := Left;
  PendCtrl := GCtrl[g];
  PendSent := 0;
  PendStart := GetTickCount;
  result := ' -> ' + Sk.Name;
  AfterTick();
end;

// an auto pick must wait now: a key pick a moment ago (your choice stays), the after skill of a pick still being sent
// (a new target would drop it), our own cast going (no target switch in the middle of it)
function AutoHeld(): boolean;
begin
  result := (GetTickCount - ManualAt < AUTO_HOLD_MS) or (PendSk <> 0) or (User.Cast.EndTime > 0);
end;

// a res-kill pick must wait now: only your key pick a moment ago. Our cast does not hold it (a new target does not
// stop the cast), nor the after skill / volley being sent (the risen enemy comes first: it is dropped)
function ResHeld(): boolean;
begin
  result := GetTickCount - ManualAt < AUTO_HOLD_MS;
end;

// group g: a key press (IsAuto false) or its auto event. true = somebody was targeted
function PickGroup(g: integer; IsAuto: boolean): boolean;
var Best, BestOID, InRange, FarI, FarOID, Cur, k: integer;
  BestD, FarD: double;
  CurIn, Fb: boolean;
  Who, Rest, Pre, Nm: string;
begin
  Fb := false;
  result := false;
  RefreshMe();
  if GMode[g] = MODE_CAST then CastScan(false);   // fresh, whole
  Cur := TargetOID();
  Pre := GName[g] + ': ';
  if IsAuto then begin
    if GMode[g] = MODE_RES then begin
      if ResHeld() then exit;          // res-kill: only your key pick a moment ago first
    end else if AutoHeld() then exit;  // your key pick / its after skill / our cast first
    Best := ScanGroup(g, Cur, false, BestOID, BestD, InRange, CurIn, FarI, FarD);
    if CurIn or (Best < 0) then exit;   // on one of them already / none in range: quiet
    Pre := GName[g] + ' (сам): ';
  end else begin
    ManualAt := GetTickCount;          // a key pick: the auto picks wait AUTO_HOLD_MS
    // a new cycle unless this group key was pressed lately and we are still on its last pick
    if (CycG <> g) or (GetTickCount - CycAt > cardinal(CycleMs)) or (CycN >= MAX_CYCLE) then CycN := 0;
    if CycN > 0 then
      if CycOID[CycN - 1] <> Cur then CycN := 0;
    Best := ScanGroup(g, Cur, true, BestOID, BestD, InRange, CurIn, FarI, FarD);
    if (Best < 0) and (CycN > 0) then begin   // all of them picked in this cycle: from the first again
      CycN := 0;
      Best := ScanGroup(g, Cur, true, BestOID, BestD, InRange, CurIn, FarI, FarD);
    end;
    if (Best < 0) and (not CurIn) and (GMode[g] = MODE_SMART) then begin   // no assister: one who targets us now
      SmartFb := true;
      Best := ScanGroup(g, Cur, true, BestOID, BestD, InRange, CurIn, FarI, FarD);
      SmartFb := false;
      Fb := Best >= 0;
    end;
    CycG := g;
    CycAt := GetTickCount;
    if Best < 0 then begin
      if CurIn then begin   // only the target now: its after action again
        if CycN = 0 then begin
          CycOID[0] := Cur;
          CycN := 1;
        end;
        Rest := DoAfter(g, Cur);
        Say(Pre + 'уже на ' + TargetName() + ', он один' + Rest);
        exit;
      end;
      CycN := 0;
      if FarI >= 0 then begin
        FarOID := 0;
        if FarI < CharList.Count then FarOID := CharList(FarI).OID;
        Rest := '';
        if FarD <= GRange[g] then Rest := ' (выше / ниже больше чем на ' + IntToStr(GHeight[g]) + ')';
        Say(Pre + 'никого в ' + IntToStr(GRange[g]) + ', ближайший ' + WhoText(FarI, FarOID) + ' - ' +
          IntToStr(Round(FarD)) + Rest);
      end else
        Say(Pre + 'никого рядом');
      exit;
    end;
  end;

  Who := WhoText(Best, BestOID);   // before the target: the list may change meanwhile
  Nm := '';
  if Best < CharList.Count then Nm := CharList(Best).Name;
  if not Engine.SetTarget(BestOID) then begin
    if GMode[g] = MODE_RES then begin
      ResNote := 'could not target ' + Nm;
      ResNoteAt := GetTickCount;
    end;
    Say(Pre + 'не взял ' + Who + ' (' + IntToStr(Round(BestD)) + ')');
    exit;
  end;
  NetTgtTick();                        // the room gets our new target now
  if GMode[g] = MODE_RES then begin    // the risen one: the after skill / volley at the old one dropped; assist /
    PendSk := 0;                       // volley leave him to us (ResKeeps)
    ResOID := BestOID;
    ResAt := GetTickCount;
    ResNote := 'took ' + Nm + ' (up ' + IntToStr(RevivedAgo(BestOID)) + ' ms)';
    ResNoteAt := GetTickCount;
  end;
  if not IsAuto then begin       // the press -> the target taken
    PfKey := GetTickCount - LastAt;
    if PfKey > PfKeyMax then PfKeyMax := PfKey;
  end;
  if not IsAuto then begin             // the cycle of a key (an auto pick leaves it as it is)
    CycG := g;
    CycAt := GetTickCount;
    CycOID[CycN] := BestOID;
    CycN := CycN + 1;
  end;
  if (GMark[g] > 0) or (GAfter[g] > 0) then Delay(50);   // the server registers the target first
  if GMark[g] > 0 then SendMark(GMark[g]);
  Rest := '';
  if GMode[g] = MODE_CAST then begin
    if GWho[g] = 0 then Rest := ' x' + IntToStr(ChargeCount(g, BestOID))
    else Rest := ' (кастует)';
  end;
  if GMode[g] = MODE_ONUS then begin
    k := FoIdx(BestOID);
    if (k >= 0) and (FoTgt[k] = MyOID) then Rest := ' (держит тебя)'
    else Rest := ' (держит пати)';
  end;
  if GMode[g] = MODE_SMART then begin
    k := FoIdx(BestOID);
    if Fb then Rest := ' (ассистеров нет - кто держит нас)'
    else if k >= 0 then Rest := ' (звал ассист ' + IntToStr(FoInit[k]) + ', за ним ' + IntToStr(FoLed[k]) + ')';
  end;
  Rest := Rest + DoAfter(g, BestOID);
  if InRange > 1 then Rest := Rest + ', в дальности ' + IntToStr(InRange);
  Say(Pre + Who + ' ' + IntToStr(Round(BestD)) + Rest);
  result := true;
end;

// OID is aggroed: an enemy player in sight holds him in his target (the followed players: numbers only)
function AggroOn(OID: integer): boolean;
var k: integer;
begin
  result := false;
  if OID = 0 then exit;
  for k := 0 to FOE_SLOTS - 1 do
    if (FoOID[k] <> 0) and (FoTgt[k] = OID) and (not FoOurs[k]) and (GetTickCount - FoSeen[k] <= FOE_SEEN_MS) then begin
      result := true;
      exit;
    end;
end;

// CharList index of OID (-1 not in sight)
function CharIdxOf(OID: integer): integer;
var i: integer;
begin
  result := -1;
  if OID = 0 then exit;
  for i := 0 to CharList.Count - 1 do if i < CharList.Count then
    if CharList(i).OID = OID then begin
      result := i;
      exit;
    end;
end;

// res-kill took him a moment ago (RES_KEEP_MS) and he is still our target, alive: assist / volley leave him to us
function ResKeeps(): boolean;
var i: integer;
begin
  result := false;
  if ResOID = 0 then exit;
  if GetTickCount - ResAt >= RES_KEEP_MS then exit;
  if TargetOID() <> ResOID then exit;              // you took another one meanwhile
  i := CharIdxOf(ResOID);
  if i < 0 then exit;
  if CharList(i).Dead then exit;
  result := true;
end;

// the room of the relay: the slot of a character (lower case), -1 none

var k: integer;
begin
  result := -1;
  for k := 0 to NtN - 1 do
    if NtName[k] = Lw then begin
      result := k;
      exit;
    end;
end;

// assist via the server, group g: its assister (the first name; none: the party leader) when his bot is in the room of
// the relay ('' none / he is me). In the party or not, dead / far / out of sight: does not matter
function NetCaller(g: integer): string;
var L: TL2Char;
  Nm: string;
begin
  result := '';
  NetSlot := -1;
  if GAsN[g] = 0 then begin
    L := Party.Leader;
    if L = nil then exit;
    if L.OID = MyOID then exit;
    Nm := LowerCase(L.Name);
  end else Nm := GAsName[g * 3];
  if Nm = LowerCase(CharName) then exit;             // he is me: I lead
  if Pos(',' + Nm + ',', NetRoom) = 0 then exit;     // his bot is not on the server
  result := Nm;
  NetSlot := 0;
end;

// the CharList index of the one group g follows via the server (in sight: his casts are seen), -1 none

  C := NetCaller(g);
  if C = '' then exit;
  for i := 0 to CharList.Count - 1 do if i < CharList.Count then

      result := i;
      exit;
    end;
end;

// an NPC / a mob of this OID in sight
function NpcSeen(OID: integer): boolean;
var i: integer;
begin
  result := false;
  for i := 0 to NpcList.Count - 1 do if i < NpcList.Count then
    if NpcList(i).OID = OID then begin
      result := true;
      exit;
    end;
end;

// CharList index of assister s (0..2) of group g: by his OID found before, by name not more often than a second (the
// names are strings of the bot: not read every look); -1 not in sight
function AssistSlot(g, s: integer): integer;
var i, j: integer;
begin
  result := -1;
  j := g * 3 + s;
  if AsOID[j] <> 0 then begin
    result := CharIdxOf(AsOID[j]);
    if result >= 0 then exit;
  end;
  if (AsFindAt[j] <> 0) and (GetTickCount - AsFindAt[j] < 1000) then exit;
  AsFindAt[j] := GetTickCount;
  for i := 0 to CharList.Count - 1 do if i < CharList.Count then
    if LowerCase(CharList(i).Name) = GAsName[j] then begin
      AsOID[j] := CharList(i).OID;
      result := i;
      exit;
    end;
end;

// whom group g assists now: the first of its list in sight, alive and (assist_aggro 2) not aggroed - aggroed / killed /
// gone -> the next one; no list -> the party leader. His CharList index (AsSlot: which), -1 none (AsWhy: why)
function AssistActive(g: integer): integer;
var s, i: integer;
  L: TL2Char;
begin
  result := -1;
  AsSlot := -1;
  AsWhy := '';
  if GAsN[g] = 0 then begin
    L := Party.Leader;
    if (L = nil) or (L.OID = MyOID) then begin
      AsWhy := 'лидера пати нет';
      exit;
    end;
    i := CharIdxOf(L.OID);
    if i < 0 then AsWhy := 'лидер пати не в видимости'
    else if CharList(i).Dead then AsWhy := 'лидер пати убит'
    else if (GAsAggro[g] = 2) and AggroOn(L.OID) then AsWhy := 'лидер пати сагрен'
    else result := i;
    exit;
  end;
  for s := 0 to GAsN[g] - 1 do begin
    if GAsName[g * 3 + s] = LowerCase(CharName) then begin   // me: only the ones above me are followed, I lead
      if s = 0 then AsWhy := 'ты первый в списке - ведёшь сам'
      else AsWhy := 'выше тебя в списке все сагрены / убиты / не видно - ведёшь сам';
      exit;
    end;
    i := AssistSlot(g, s);
    if i < 0 then continue;                          // not in sight
    if CharList(i).Dead then continue;               // killed
    if (GAsAggro[g] = 2) and AggroOn(CharList(i).OID) then continue;   // aggroed
    result := i;
    AsSlot := s;
    exit;
  end;
  AsWhy := 'все ассистеры сагрены / убиты / не видно';
end;

// assist group g: the target of the player it assists (not one of ours, not me) + the after action; not while the
// aggro check says aggroed (assist_aggro: 2 the assisted player, 1 me). true = taken
function AssistPick(g: integer; IsAuto: boolean): boolean;
var i, OID, k, HisOID: integer;
  TL: TL2Live;
  Who, Rest, Pre, His: string;
begin
  result := false;
  RefreshMe();
  Pre := GName[g] + ': ';
  if IsAuto then Pre := GName[g] + ' (сам): ';
  if (GAsAggro[g] = 1) and AggroOn(MyOID) then begin
    if not IsAuto then Say(Pre + 'не ассистирую: ты сагрен');
    exit;
  end;
  i := AssistActive(g);
  if i < 0 then begin
    if not IsAuto then Say(Pre + AsWhy);
    exit;
  end;
  HisOID := CharList(i).OID;
  OID := 0;
  TL := CharList(i).Target;
  if TL <> nil then OID := TL.OID;
  if (OID = 0) or (OID = MyOID) then begin
    if not IsAuto then Say(Pre + 'у него нет цели');
    exit;
  end;
  k := CharIdxOf(OID);
  if (k >= 0) and IsOurs(k) then begin
    if not IsAuto then Say(Pre + 'его цель - свой');
    exit;
  end;
  if OID = TargetOID() then begin           // on his target already: the after action again (a key)
    if not IsAuto then begin
      Rest := DoAfter(g, OID);
      Say(Pre + 'уже на его цели' + Rest);
    end;
    exit;
  end;
  if IsAuto then begin
    if AutoHeld() then exit;
    if ResKeeps() then exit;                 // the res-kill target first
  end else ManualAt := GetTickCount;
  His := '';
  if i < CharList.Count then His := CharList(i).Name;
  if k >= 0 then Who := WhoText(k, OID) else Who := 'его цель (не игрок)';
  if not Engine.SetTarget(OID) then begin
    Say(Pre + 'не взял ' + Who);
    exit;
  end;
  NetTgtTick();
  if (GMark[g] > 0) or (GAfter[g] > 0) then Delay(50);   // the server registers the target first
  if GMark[g] > 0 then SendMark(GMark[g]);
  Rest := DoAfter(g, OID);
  if AsSlot > 0 then His := His + ', ' + IntToStr(AsSlot + 1) + '-й: первые сагрены / убиты / не видно';
  Say(Pre + Who + ' (цель ' + His + ')' + Rest);
  result := true;
end;

// assist groups that are on (their key / auto=1): his target changed -> taken (when the aggro check lets; else later)
procedure AssistTick();
var g, i, OID: integer;
  TL: TL2Live;
begin
  for g := 0 to GroupN - 1 do begin
    if (GMode[g] <> MODE_ASSIST) or (not GResOn[g]) or GAsNet[g] then continue;   // via the server: NetTick
    i := AssistActive(g);                    // the first one usable now: the next one when he is aggroed / killed
    if i < 0 then continue;
    OID := 0;
    TL := CharList(i).Target;
    if TL <> nil then OID := TL.OID;
    if OID = AsLast[g] then continue;       // no change
    if (OID = 0) or (OID = MyOID) or (OID = TargetOID()) then begin
      AsLast[g] := OID;
      continue;
    end;
    if AssistPick(g, true) then AsLast[g] := OID
    else if (CharIdxOf(OID) >= 0) and IsOurs(CharIdxOf(OID)) then AsLast[g] := OID;   // a friend: nothing to wait for
  end;
end;

// a volley (an assist group on, volley=): the one it follows started one of its skills -> his target + the same skill
// of ours (sent till it starts, as the after skill). Ours not there / in its reuse -> nothing
procedure VolleyTick();
var g, i, k, Tg, Left: integer;
  Sk: TL2Skill;
begin
  g := VolG;
  if g < 0 then exit;
  VolG := -1;
  if (g >= GroupN) or (GMode[g] <> MODE_ASSIST) or (not GResOn[g]) then exit;
  if GAsNet[g] then i := NetCallerIdx(g) else i := AssistActive(g);
  if i < 0 then exit;
  if CharList(i).OID <> VolOID then exit;            // another one of the list: not the one we follow now
  Tg := VolTgt;
  if (Tg = 0) or (Tg = MyOID) then exit;
  if (Tg <> ResOID) and (not GAsNet[g]) then
    if ResKeeps() then begin                         // we are on the one res-kill took: he first
      Engine.Msg('[PvP]', GName[g] + ': volley skipped - the res-kill target first', 0);
      exit;
    end;
  k := CharIdxOf(Tg);
  if k >= 0 then
    if IsOurs(k) then exit;                          // his skill at a friend
  if not SkillList.ByID(VolSk, Sk) then exit;        // we have no such skill
  Left := Sk.EndTime;
  if Left > 1000 then begin
    Engine.Msg('[PvP]', GName[g] + ': volley ' + Sk.Name + ' - ours in its reuse', 0);
    exit;
  end;
  if TargetOID() <> Tg then
    if not Engine.SetTarget(Tg) then exit;
  AsLast[g] := Tg;                                   // his target is ours: AssistTick does not take it again
  PendSk := VolSk;
  PendTgt := Tg;
  PendL0 := Left;
  PendCtrl := GCtrl[g];
  PendSent := 0;
  PendStart := GetTickCount;
  AfterTick();
  Engine.Msg('[PvP]', GName[g] + ': volley ' + Sk.Name, 0);
end;

// assist via the server (groups on, assist_src=net): the assister's bot sent a new target -> taken as it is (no range,
// no aggro, no holds) + the after action (not at a friend: selected only). It only waits till the game knows that
// target (in sight); SetTarget failed -> again in NET_TRY_MS. Never me
procedure NetTick();
var g, k, i, W: integer;
  C, Who, Rest: string;
begin
  NetDirty := false;
  for g := 0 to GroupN - 1 do begin
    if (GMode[g] <> MODE_ASSIST) or (not GAsNet[g]) or (not GResOn[g]) then continue;
    C := NetCaller(g);
    if C = '' then continue;                         // he is not on the server / he is me
    k := NetIdx(C);
    if k < 0 then continue;                          // no target from him yet
    W := NtOID[k];
    if W = NetTook[g] then continue;                 // done already
    if W <> NetWant[g] then begin                    // a new one: tried at once
      NetWant[g] := W;
      NetTryAt[g] := 0;
    end;
    if (W = 0) or (W = MyOID) or (W = TargetOID()) then begin
      NetTook[g] := W;
      AsLast[g] := W;
      continue;
    end;
    i := CharIdxOf(W);
    if i < 0 then
      if not NpcSeen(W) then continue;               // not in sight yet: taken as soon as he is
    if GetTickCount - NetTryAt[g] < NET_TRY_MS then continue;
    NetTryAt[g] := GetTickCount;
    if i >= 0 then Who := WhoText(i, W) else Who := NtTName[k];
    PendSk := 0;                                     // an after skill / volley at the old one: dropped
    if not Engine.SetTarget(W) then continue;        // tried again
    NetTgtTick();
    NetTook[g] := W;
    AsLast[g] := W;
    i := CharIdxOf(W);
    if i >= 0 then
      if IsOurs(i) then begin                        // a friend: selected, no attack at him
        Say(GName[g] + ' (сервер): ' + Who + ' - свой, без атаки');
        continue;
      end;
    if (GMark[g] > 0) or (GAfter[g] > 0) then Delay(50);   // the server registers the target first
    if GMark[g] > 0 then SendMark(GMark[g]);
    Rest := DoAfter(g, W);
    Say(GName[g] + ' (сервер): ' + Who + ' (цель ' + NtShow[k] + ')' + Rest);
  end;
end;

// a successful res-kill (a res-kill group on with kill_sound=): an enemy up from the dead dies again within its
// kill_ms, not seen rebuffed meanwhile (KILL_BUFFED) -> the sound + a notice. Each revive looked at once
procedure KillSoundTick();
var k, i, g, Snd, n, Rk: integer;
  SndPath: string;
begin
  Snd := -1;
  for g := 0 to GroupN - 1 do
    if (Snd < 0) and (GMode[g] = MODE_RES) and GResOn[g] and (GKillSnd[g] <> '') then Snd := g;
  if Snd < 0 then exit;
  for k := 0 to REV_SLOTS - 1 do begin
    if (RevOID[k] = 0) or RevDone[k] then continue;
    if GetTickCount - RevAt[k] > cardinal(GKillMs[Snd]) then begin   // he lived on: no res-kill
      RevDone[k] := true;
      continue;
    end;
    i := CharIdxOf(RevOID[k]);
    if i < 0 then continue;
    if not CharList(i).Dead then begin
      n := 0;
      try
        n := CharList(i).Buffs.Count;
      except
        n := 0;
      end;
      if n >= KILL_BUFFED then RevBuf[k] := true;
      continue;
    end;
    RevDone[k] := true;                              // dead again: looked at once
    if RevBuf[k] or IsOurs(i) then continue;
    if PlayerTag(LowerCase(CharList(i).Name), Rk) = 2 then continue;   // never touched
    if (PlayerTag(LowerCase(CharList(i).Name), Rk) <> 1) and (not IsFoeClan(i)) then continue;   // not an enemy
    SndPath := Script.Path + SOUND_DIR + GKillSnd[Snd];
    if FileExists(SndPath) then PlaySound(SndPath, false)
    else Engine.Msg('[PvP]', 'no sound file ' + SndPath, 0);
    Say('Рескилл: ' + CharList(i).Name + '!');
  end;
end;

// res-kill groups that are on, right after each look at who got up (the window is short): somebody up from the dead
// in the window -> taken + the after action. Your key pick a moment ago holds it: it waits then (the window stretched
// by the wait), at most RES_LATE_MS. Then the sound of a res-kill that worked
procedure ResTick();
var g: integer;
begin
  for g := 0 to GroupN - 1 do begin
    if (GMode[g] <> MODE_RES) or (not GResOn[g]) then begin
      ResHeldAt[g] := 0;
      continue;
    end;
    if GetTickCount - AutoAt[g] < cardinal(GAutoMs[g]) then continue;   // its pace
    if ResHeldAt[g] = 0 then
      if ResPending(g) then ResHeldAt[g] := GetTickCount;
    if ResHeldAt[g] = 0 then continue;
    if ResHeld() then begin
      if GetTickCount - ResHeldAt[g] > RES_LATE_MS then begin
        ResHeldAt[g] := 0;
        ResNote := 'let go: you kept picking by keys';
        ResNoteAt := GetTickCount;
        Engine.Msg('[PvP]', GName[g] + ': ' + ResNote, 0);
      end;
      continue;
    end;
    ResGrace := GetTickCount - ResHeldAt[g] + REV_SCAN_MS;
    if PickGroup(g, true) then AutoAt[g] := GetTickCount;
    ResGrace := 0;
    ResHeldAt[g] := 0;
  end;
  KillSoundTick();
end;

// ===================== CP potions =====================

// a thread of its own: our CP below CpPct % -> the potion (not sooner than CpGap). Numbers only (no strings: two
// threads building strings at once crash PaxScript); the potions left are looked at by the main loop (CpCheck)
procedure CpPots_Thread();
var Cur, Mx: integer;
begin
  while true do begin
    Delay(CP_LOOK_MS);
    if (CpItem <= 0) or (not CpHave) then continue;
    if User.Dead then continue;
    Mx := User.MaxCP;
    if Mx <= 0 then continue;
    Cur := User.CurCP;
    if Cur * 100 >= Mx * CpPct then continue;
    if GetTickCount - CpAt < cardinal(CpGap) then continue;
    CpAt := GetTickCount;
    Engine.UseItem(CpItem);
    Inc(CpUsed);
  end;
end;

// the potions left (main loop, every second): none -> the thread waits, told once
procedure CpCheck();
var It: TL2Item;
  Have: boolean;
begin
  if CpItem <= 0 then begin
    CpHave := false;
    exit;
  end;
  Have := Inventory.User.ByID(CpItem, It);
  if Have then Have := It.Count > 0;
  if (not Have) and (not CpTold) then begin
    CpTold := true;
    Say('CP-банок нет (ID ' + IntToStr(CpItem) + ')');
  end;
  if Have then CpTold := false;
  CpHave := Have;
end;

// the presses waiting: done now (between the scans too - a scan stopped for a press does not keep it waiting)
procedure HandleKeys();
var g, j: integer;
  S, C: string;
begin
  while NextG >= 0 do begin
    g := NextG;
    NextG := -1;
    if g >= GroupN then continue;
    if GMode[g] = MODE_RES then begin            // res-kill: its key turns it on / off
      GResOn[g] := not GResOn[g];
      AutoAt[g] := 0;
      if GResOn[g] then Say(GName[g] + ': ВКЛ (встал ' + IntToStr(GResMin[g]) + '-' + IntToStr(GResMax[g]) + ' мс назад)')
      else Say(GName[g] + ': ВЫКЛ');
    end else if GMode[g] = MODE_ASSIST then begin   // assist: its key turns the following on / off
      GResOn[g] := not GResOn[g];
      AsLast[g] := 0;                                 // on: his target now is taken at once (AssistTick)
      NetTook[g] := -1;                               // via the server: the same (NetTick)
      NetWant[g] := -1;
      NetDirty := true;
      if GResOn[g] then begin
        S := '';
        for j := 1 to Length(GAssist[g]) do begin
          C := Copy(GAssist[g], j, 1);
          if C = '|' then S := S + ', ' else S := S + C;
        end;
        if S = '' then S := 'лидер пати';
        if GAsNet[g] then begin                       // via the server: one assister
          j := Pos(',', S);
          if j > 0 then S := Copy(S, 1, j - 1);
          S := 'через сервер, ' + S;
        end;
        Say(GName[g] + ': ВКЛ (ассист ' + S + ')');
      end else Say(GName[g] + ': ВЫКЛ');
    end else PickGroup(g, false);
  end;
end;

// the whole file, shared with a writer (the page); false = not there / not read
function ReadText(Path: string; L: TStringList): boolean;
var FS: TFileStream;
begin
  result := false;
  L.Clear;
  if not FileExists(Path) then exit;   // no exception for that: the bot prints even a caught one
  try
    FS := TFileStream.Create(Path, fmOpenRead or fmShareDenyNone);
    try
      L.LoadFromStream(FS);
      result := true;
    finally
      FS.Free;
    end;
  except
    result := false;
  end;
end;

// the file, shared with readers (the page reads it every second)
procedure WriteText(Path: string; L: TStringList; var Told: boolean);
var FS: TFileStream;
  Ok: boolean;
begin
  Ok := true;
  try
    FS := TFileStream.Create(Path, fmCreate or fmShareDenyNone);
    try
      L.SaveToStream(FS);
    finally
      FS.Free;
    end;
  except
    Ok := false;
  end;
  if (not Ok) and (not Told) then begin
    Told := true;
    Engine.Msg('[PvP]', 'Cannot write ' + Path + ' - is there the folder Library\PvP\profiles?', 128);
  end;
end;

function ModeOf(S: string): integer;
begin
  S := LowerCase(Trim(S));
  if S = 'class' then result := MODE_CLASS
  else if S = 'reskill' then result := MODE_RES
  else if S = 'cast' then result := MODE_CAST
  else if S = 'names' then result := MODE_NAMES
  else if S = 'smart' then result := MODE_SMART
  else if S = 'onus' then result := MODE_ONUS
  else if S = 'assist' then result := MODE_ASSIST
  else result := -1;
end;

// 'classes=97:1|16:1|103:2' (class:priority) into group n
procedure SetClasses(n: integer; Txt: string; P: TStringList);
var j, c, ID, T: integer;
  S: string;
begin
  GClsN[n] := 0;
  SplitTo(Txt, '|', P);
  for j := 0 to P.Count - 1 do begin
    S := Trim(P[j]);
    c := Pos(':', S);
    T := 1;
    if c > 0 then begin
      T := StrToIntDef(Copy(S, c + 1, 3), 1);
      S := Copy(S, 1, c - 1);
    end;
    ID := StrToIntDef(S, -1);
    if (ID < 0) or (GClsN[n] >= MAX_CLS) then continue;
    if T < 1 then T := 1;
    if T > 9 then T := 9;
    GCls[n * MAX_CLS + GClsN[n]] := ID;
    GTier[n * MAX_CLS + GClsN[n]] := T;
    GClsN[n] := GClsN[n] + 1;
  end;
end;

// 'skills=426|427' into group n
procedure SetSkills(n: integer; Txt: string; P: TStringList);
var j, ID: integer;
begin

  for j := 0 to P.Count - 1 do begin
    ID := StrToIntDef(Trim(P[j]), 0);
    if (ID <= 0) or (GSkN[n] >= MAX_SK) then continue;
    GSk[n * MAX_SK + GSkN[n]] := ID;
    GSkN[n] := GSkN[n] + 1;
  end;
end;

  end;
end;

// 'after=' / 'after=attack' / 'after=skill:<ID>'
procedure SetAfter(n: integer; Val: string);
begin
  GAfter[n] := 0;
  GAfterSk[n] := 0;
  Val := LowerCase(Trim(Val));
  if Val = 'attack' then GAfter[n] := 1
  else if Copy(Val, 1, 6) = 'skill:' then begin
    GAfter[n] := 2;
    GAfterSk[n] := StrToIntDef(Copy(Val, 7, 12), 0);
  end;
end;

// 'clans=One|Two' -> ClanKey / ClanText
procedure SetClans(Txt: string; P: TStringList);
var j: integer;
  S: string;
begin
  ClanKey := '';
  ClanText := '';
  SplitTo(Txt, '|', P);
  for j := 0 to P.Count - 1 do begin
    S := Trim(P[j]);
    if S = '' then continue;
    if ClanKey = '' then ClanKey := ',';
    ClanKey := ClanKey + LowerCase(S) + ',';
    if ClanText <> '' then ClanText := ClanText + ', ';
    ClanText := ClanText + S;

end;

// 'players=+Rhea|+Bourbon|-Friend' (+ priority in this order, - never)
procedure SetPlayers(Txt: string; P: TStringList);
var j, R: integer;
  S, C: string;
begin
  PlN := 0;
  R := 0;
  SplitTo(Txt, '|', P);
  for j := 0 to P.Count - 1 do begin
    S := Trim(P[j]);
    if (Length(S) < 2) or (PlN >= MAX_PL) then continue;
    C := Copy(S, 1, 1);
    if (C <> '+') and (C <> '-') then continue;
    PlName[PlN] := LowerCase(Trim(Copy(S, 2, Length(S))));
    if C = '+' then begin
      PlTag[PlN] := 1;
      PlRank[PlN] := R;
      R := R + 1;
    end else begin
      PlTag[PlN] := 2;
      PlRank[PlN] := 99;
    end;
    PlN := PlN + 1;
  end;
end;

// the profile file (its lines in L): the general part, then a [group] block per group
procedure LoadProfile(L: TStringList);
var P: TStringList;
  i, j, n, k, p0, Code, Md: integer;
  S, Fld, Val, Clans, Players: string;
  Ok, AutoOk: boolean;
begin
  P := TStringList.Create;
  CfgErrJson := '';
  KeepN := 0;        // the res-kill / assist groups on / off now: kept over the reading (by name)
  for i := 0 to GroupN - 1 do
    if ((GMode[i] = MODE_RES) or (GMode[i] = MODE_ASSIST)) and (KeepN < 16) then begin
      KeepName[KeepN] := LowerCase(GName[i]);
      KeepOn[KeepN] := GResOn[i];
      Inc(KeepN);
    end;
  GroupN := 0;       // the key thread sees none meanwhile
  NextG := -1;
  CycN := 0;
  PendSk := 0;
  VolG := -1;
  ProfName := ProfId;
  ChatOn := true;
  ChatName := 'PvP';
  NotifyType := 15;
  CpItem := 5592;                                    // Greater CP Potion (a profile without cp_item: on)
  CpPct := 95;
  CpGap := 150;
  CycleMs := 6000;
  ProfRange := 1200;
  Clans := '';
  Players := '';
  n := -1;   // the group being read; -1: the general part
  for i := 0 to L.Count - 1 do begin
    S := Trim(L[i]);
    if S = '' then continue;
    if (Copy(S, 1, 1) = ';') or (Copy(S, 1, 1) = '#') then continue;
    if LowerCase(S) = '[group]' then begin
      if n + 1 >= MAX_GROUPS then begin
        CfgErr('More than ' + IntToStr(MAX_GROUPS) + ' groups - the rest skipped');
        break;
      end;
      n := n + 1;
      ResetGroup(n);
      continue;
    end;
    p0 := Pos('=', S);
    if p0 = 0 then continue;
    Fld := LowerCase(Trim(Copy(S, 1, p0 - 1)));
    Val := Trim(Copy(S, p0 + 1, Length(S)));
    if n < 0 then begin
      if (Fld = 'name') and (Val <> '') then ProfName := Val
      else if Fld = 'clans' then Clans := Val
      else if Fld = 'players' then Players := Val
      else if Fld = 'chat' then begin
        ChatOn := Val <> '0';
        if not ChatOn then NotifyType := 0;
      end
      else if Fld = 'notify' then NotifyType := StrToIntDef(Val, 15)
      else if Fld = 'cp_item' then CpItem := StrToIntDef(Val, 0)
      else if Fld = 'range' then ProfRange := StrToIntDef(Val, 1200)
      else if Fld = 'cp_pct' then CpPct := StrToIntDef(Val, 95)
      else if Fld = 'cp_gap_ms' then CpGap := StrToIntDef(Val, 150)
      else if (Fld = 'chat_name') and (Val <> '') then ChatName := Val
      else if Fld = 'cycle_ms' then CycleMs := StrToIntDef(Val, 6000);
      continue;
    end;
    if Fld = 'on' then GOn[n] := Val <> '0'
    else if (Fld = 'name') and (Val <> '') then GName[n] := Val
    else if Fld = 'key' then GKeyText[n] := Val
    else if Fld = 'mode' then begin
      Md := ModeOf(Val);
      if Md < 0 then CfgErr(GName[n] + ': unknown mode "' + Val + '" - classes taken')
      else GMode[n] := Md;
    end
    else if Fld = 'range' then GRange[n] := StrToIntDef(Val, 0)
    else if Fld = 'height' then GHeight[n] := StrToIntDef(Val, 400)
    else if Fld = 'mark' then GMark[n] := StrToIntDef(Val, 0)
    else if Fld = 'after' then SetAfter(n, Val)
    else if Fld = 'ctrl' then GCtrl[n] := Val <> '0'
    else if Fld = 'auto' then GAuto[n] := Val = '1'
    else if Fld = 'auto_ms' then GAutoMs[n] := StrToIntDef(Val, 3000)
    else if Fld = 'window' then GWindow[n] := StrToIntDef(Val, 15)
    else if Fld = 'res_min_ms' then GResMin[n] := StrToIntDef(Val, 0)
    else if Fld = 'res_max_ms' then GResMax[n] := StrToIntDef(Val, -1)
    else if Fld = 'assist' then GAssist[n] := Val
    else if Fld = 'assist_aggro' then GAsAggro[n] := StrToIntDef(Val, 2)
    else if Fld = 'assist_src' then GAsNet[n] := LowerCase(Trim(Val)) = 'net'
    else if Fld = 'who' then begin
      if LowerCase(Val) = 'caster' then GWho[n] := 1 else GWho[n] := 0;
    end
    else if Fld = 'prio_first' then GPrio[n] := Val <> '0'
    else if Fld = 'classes' then SetClasses(n, Val, P)
    else if Fld = 'volley' then SetVolley(n, Val, P)
    else if Fld = 'kill_sound' then GKillSnd[n] := Val

    Code := ParseKey(GKeyText[i]);

      else begin
        CfgErr(GName[i] + ': cannot read the key "' + GKeyText[i] + '" - skipped');
        continue;
      end;
    end;
    if (Code = 0) and (not AutoOk) then begin
      CfgErr(GName[i] + ': no key - skipped');
      continue;
    end;
    Ok := true;
    if Code <> 0 then
      for j := 0 to k - 1 do
        if GKey[j] = Code then begin
          CfgErr(GName[i] + ': the key ' + GKeyText[i] + ' is taken by ' + GName[j] + ' - skipped');

    end;

    if GAutoMs[i] < 500 then GAutoMs[i] := 500;

    NetWant[k] := -1;

    SplitTo(GAssist[k], '|', P);
    for j := 0 to P.Count - 1 do
      if (Trim(P[j]) <> '') and (GAsN[k] < 3) then begin
        GAsName[k * 3 + GAsN[k]] := LowerCase(Trim(P[j]));
        AsOID[k * 3 + GAsN[k]] := 0;
        AsFindAt[k * 3 + GAsN[k]] := 0;
        GAsN[k] := GAsN[k] + 1;
      end;
    // res-kill / assist: on at the start if auto=1 (its key turns it on / off)
    GResOn[k] := ((GMode[k] = MODE_RES) or (GMode[k] = MODE_ASSIST)) and GAuto[k];
    if (GMode[k] = MODE_RES) or (GMode[k] = MODE_ASSIST) then
      for j := 0 to KeepN - 1 do
        if KeepName[j] = LowerCase(GName[k]) then GResOn[k] := KeepOn[j];   // as it was before the reading
    k := k + 1;
  end;
  GroupN := k;
  P.Free;
end;

// which profile: 'active=<id>' of pvp.ini (every character without his own), else 'main'
function PickProfile(L: TStringList; var How: string): string;
var i, p0: integer;
  S, Fld, Val, Act: string;
begin
  Act := '';
  for i := 0 to L.Count - 1 do begin
    S := Trim(L[i]);
    if Copy(S, 1, 1) = ';' then continue;
    p0 := Pos('=', S);
    if p0 = 0 then continue;
    Fld := LowerCase(Trim(Copy(S, 1, p0 - 1)));
    Val := Trim(Copy(S, p0 + 1, Length(S)));
    if Fld = 'active' then Act := Val;
  end;
  if Act <> '' then begin
    result := FileTag(Act);
    How := 'active';
  end else begin
    result := 'main';
    How := 'default';
  end;
end;

// ===================== the PvP server (pvp_server.py): the page live =====================
// The relay (assist via the server) goes through it as well: ours N|T|<oid>|<name> (our target changed); its
// N|LINK|1|<host> / N|LINK|0|<why>, N|ROOM|a,b,c, N|T|<char>|<oid>|<name>, N|RTT|<ms>, K (read: nothing to do).
// One line per message (CRLF). Ours: HELLO|<character>, STATE|<json>, PONG. Its: RELOAD (the page saved: read the
// files now), PING. No server: tried again every SRV_RETRY_MS - the keys work all the same (the files)

procedure SrvConnect();
begin
  SrvAt := GetTickCount;
  Srv := TTCPBlockSocket.Create;
  Srv.Connect(SRV_HOST, SRV_PORT);
  if Srv.LastError <> 0 then begin
    SrvClose();
    if not SrvTold then begin
      SrvTold := true;
      Engine.Msg('[PvP]', 'No PvP server: the keys work, the page is not live - start Library\PvP\pvp_server.bat ' +
        '(then ' + SRV_PAGE + ')', 0);
    end;
    exit;
  end;
  SrvOn := true;
  Watching := true;                                  // till the server says (an old one never does)
  SrvSend('HELLO|' + CharName);
  NetSent := -1;                                     // our target: sent again
  if SrvOn then begin
    SrvTold := false;
    Engine.Msg('[PvP]', 'PvP server: linked - the page ' + SRV_PAGE, 0);
  end;
end;

// the server is gone (an error other than "nothing came"): tried again later
procedure SrvLost(Er: integer);
begin
  SrvClose();
  SrvAt := GetTickCount;
  Engine.Msg('[PvP]', 'PvP server: the link is lost (' + IntToStr(Er) + ') - trying again', 0);
end;

// a line of the relay (through PvpPanel): N|LINK|1|<host> / N|LINK|0|<why>, N|ROOM|a,b,c, N|T|<char>|<oid>|<name>,
// N|RTT|<ms>
procedure NetLine(Line: string);
var P: TStringList;
  k, j: integer;
  Kind, Lw: string;
begin
  P := TStringList.Create;
  SplitTo(Line, '|', P);
  if P.Count >= 2 then begin
    Kind := P[1];
    if (Kind = 'LINK') and (P.Count >= 3) then begin
      NetLink := StrToIntDef(P[2], 0);
      NetWhy := '';
      if P.Count >= 4 then NetWhy := P[3];
    end else if Kind = 'ROOM' then begin
      NetRoomShow := '';
      if P.Count >= 3 then NetRoomShow := P[2];
      NetRoom := ',' + LowerCase(NetRoomShow) + ',';
      if NetLink < 0 then NetLink := 0;
      NetDirty := true;
    end else if (Kind = 'T') and (P.Count >= 4) then begin
      Lw := LowerCase(Trim(P[2]));
      k := NetIdx(Lw);
      if k < 0 then begin
        if NtN < NET_SLOTS then begin
          k := NtN;
          NtN := NtN + 1;
        end else begin                               // full: the oldest one goes
          k := 0;
          for j := 1 to NtN - 1 do
            if NtAt[j] < NtAt[k] then k := j;
        end;
      end;
      NtName[k] := Lw;
      NtShow[k] := Trim(P[2]);
      NtOID[k] := StrToIntDef(P[3], 0);
      NtTName[k] := '';
      if P.Count >= 5 then NtTName[k] := P[4];
      NtAt[k] := GetTickCount;
      NetDirty := true;
    end else if (Kind = 'RTT') and (P.Count >= 3) then NetRtt := StrToIntDef(P[2], -1);
  end;
  P.Free;
end;

// one line of the server
procedure SrvLine(Line: string);
begin
  if Line = '' then exit;
  if Copy(Line, 1, 2) = 'N|' then NetLine(Line)    // the relay
  else if Line = 'RELOAD' then CfgAt := 0           // the page saved: the files read now
  else if Line = 'PING' then SrvSend('PONG')
  else if Line = 'STATRESET' then begin            // the page: the stats from zero
    FoResetAll();
    FoesAt := 0;
    Engine.Msg('[PvP]', 'the stats are reset', 0);
  end
  else if Copy(Line, 1, 6) = 'WATCH|' then begin   // a page watches / none any more
    Watching := Copy(Line, 7, 1) <> '0';
    if Watching then StateAt := 0;                 // the state now
  end;
end;

// the server's lines (a few per look, never waiting); no link -> a new try now and then
procedure SrvPoll();
var Line: string;
  n, Er: integer;
begin
  if not SrvOn then begin
    if GetTickCount - SrvAt >= SRV_RETRY_MS then SrvConnect();
    exit;
  end;
  n := 0;
  while n < 10 do begin
    Line := Srv.RecvString(1);
    Er := Srv.LastError;
    if Er <> 0 then begin
      if Er <> 10060 then SrvLost(Er);   // not "nothing came": the server is gone
      exit;
    end;
    Inc(n);
    SrvLine(Trim(Line));
  end;
end;

// the pause of the main loop (Ms), in NET_STEP_MS steps: the server's lines read as they come (waiting on the link,
// not sleeping), our target sent as soon as it changes; a target / the room from the relay or a key press ends it
procedure SrvWait(Ms: integer);
var T0: cardinal;
  Step, Er: integer;
  Line: string;
begin
  T0 := GetTickCount;
  while true do begin
    NetTgtTick();
    if NetDirty then exit;
    if KeyPending() then exit;
    Step := Ms - integer(GetTickCount - T0);
    if Step <= 0 then exit;
    if Step > NET_STEP_MS then Step := NET_STEP_MS;
    if SrvOn then begin
      Line := Srv.RecvString(Step);
      Er := Srv.LastError;
      if Er = 0 then SrvLine(Trim(Line))
      else if Er <> 10060 then SrvLost(Er);
    end else Delay(Step);
  end;
end;

// our skills that are not passive, as JSON (built anew only when they change)
function SkillsJson(): string;
var i, n, Sig: integer;
  Sk: TL2Skill;
  S: string;
begin
  Sig := SkillList.Count;
  for i := 0 to SkillList.Count - 1 do if i < SkillList.Count then begin
    Sk := SkillList.Items(i);
    if Sk = nil then continue;
    Sig := (Sig * 31 + Sk.ID * 13 + Sk.Level) and $3FFFFFF;
  end;
  if (Sig = SkSig) and (SkJson <> '') then begin
    result := SkJson;
    exit;
  end;
  SkSig := Sig;
  S := '';
  n := 0;
  for i := 0 to SkillList.Count - 1 do if i < SkillList.Count then begin
    Sk := SkillList.Items(i);
    if Sk = nil then continue;
    if Sk.Passive then continue;
    if n > 0 then S := S + ', ';
    S := S + '{"id": ' + IntToStr(Sk.ID) + ', "lvl": ' + IntToStr(Sk.Level) + ', "name": ' + JsonStr(Sk.Name) + '}';
    n := n + 1;
  end;
  SkJson := '[' + S + ']';
  result := SkJson;
end;

  JAdd('{"char": ' + JsonStr(CharName) + ', "written": ' + JsonStr(FormatDateTime('yyyy-mm-dd hh:nn:ss', Now)) + ',');
  JAdd(' "profile": {"id": ' + JsonStr(ProfId) + ', "name": ' + JsonStr(ProfName) + ', "how": ' + JsonStr(SelHow) +
    ', "read": ' + JsonStr(CfgReadAt) + ', "groups": ' + IntToStr(GroupN) + ', "live": [' + S + ', "errors": [' + CfgErrJson + ']},');
  JAdd(' "me": {"oid": ' + IntToStr(MyOID) + ', "clan": ' + JsonStr(Cl) + ', "ally": ' + JsonStr(Al) +
    ', "target": ' + IntToStr(TargetOID()) + ', "dead": ' + DeadText + '},');
  JAdd(' "perf": {"key": ' + IntToStr(PfKey) + ', "keymax": ' + IntToStr(PfKeyMax) + ', "dump": ' + IntToStr(PfDump) +
    ', "cast": ' + IntToStr(PfCast) + ', "loop": ' + IntToStr(PfLoop) + '},');
  JAdd(' "net": ' + NetJson() + ',');
  JAdd(' "party": ' + PartyJson() + ',');
  JAdd(' "skills": ' + SkillsJson() + ',');
  if (FoesAt = 0) or (GetTickCount - FoesAt >= FOES_MS) then begin   // the stats: once a second is enough
    FoesAt := GetTickCount;
    FoesJson();
  end;
  JAdd(' "casts": [');
  for k := 0 to SnN - 1 do begin
    if k = 0 then S := '  ' else S := ' ,';
    JAdd(S + '{"id": ' + IntToStr(SnSk[k]) + ', "n": ' + IntToStr(SnCnt[k]) + ', "k": ' + IntToStr(SnCls[k]) +
      ', "ago": ' + IntToStr((GetTickCount - SnAt[k]) div 1000) + '}');
  end;
  JAdd(' ],');
  JAdd(' "near": [');
  n := 0;
  for i := 0 to CharList.Count - 1 do if i < CharList.Count then begin
    if n >= MAX_NEAR then break;
    if (i mod SCAN_CHUNK = SCAN_CHUNK - 1) and KeyPending() then begin   // the key first: the state the next time
      StOut := '';
      exit;
    end;
    OID := CharList(i).OID;
    if (OID = 0) or (OID = MyOID) then continue;
    PX := CharList(i).X;
    PY := CharList(i).Y;
    PZ := CharList(i).Z;
    D := User.DistTo(PX, PY, PZ);
    if D > NEAR_RANGE then continue;
    Cls := CharList(i).ClassID;
    F := 0;
    if CharList(i).IsMember then F := 1
    else if IsOurs(i) then F := 4;
    if CharList(i).Dead then F := F + 2;
    S := '{"o": ' + IntToStr(OID) + ', ' + NameJson(i, OID) + ', "k": ' + IntToStr(Cls) + ', "d": ' + IntToStr(Round(D)) + ', "x": ' + IntToStr(PX - UX) +
      ', "y": ' + IntToStr(PY - UY) + ', "h": ' + IntToStr(PZ - UZ) + ', "f": ' + IntToStr(F);
    k := CastIdx(OID);
    if k >= 0 then S := S + ', "s": ' + IntToStr(CsSk[k]) + ', "t": ' + IntToStr(CsTgt[k]);
    k := FoIdx(OID);
    if (k >= 0) and (FoTgt[k] <> 0) then S := S + ', "g": ' + IntToStr(FoTgt[k]);
    Ago := RevivedAgo(OID);
    if Ago < 120000 then S := S + ', "rv": ' + IntToStr(Ago);
    if n = 0 then S := '  ' + S else S := ' ,' + S;
    JAdd(S + '}');
    n := n + 1;
  end;
  JAdd(' ]}');
  SrvSend('STATE|' + StOut);
  StOut := '';
  if GetTickCount - T0 > cardinal(PfDump) then PfDump := GetTickCount - T0;
end;
