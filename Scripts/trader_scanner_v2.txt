// ===== Helpers (no inline) =====

function SqrI(a: Integer): Int64;
begin
  Result := Int64(a) * a;
end;

function DistanceSq(ax, ay, az, bx, by, bz: Integer): Int64;
begin
  Result := SqrI(ax - bx) + SqrI(ay - by) + SqrI(az - bz);
end;

procedure SafeSleep(ms: Cardinal);
begin
  Delay(ms);
end;

function RandBetween(a, b: Cardinal): Cardinal;
begin
  Result := a + Cardinal(Random(Integer(b - a + 1)));
end;

procedure TrySendTradeShout();
begin
  if (GetTickCount - shoutLastTs) < shoutIntervalMs then Exit;

  Engine.Say(SHOUT_MESSAGES[Low(SHOUT_MESSAGES) + Random(High(SHOUT_MESSAGES) - Low(SHOUT_MESSAGES) + 1)], 1, '');
  shoutLastTs := GetTickCount;
  shoutIntervalMs := RandBetween(120000, 420000);
end;

function PickRandomMap(): string;
var
  idx: Integer;
begin
  idx := Random(High(MAP_LIST) - Low(MAP_LIST) + 1) + Low(MAP_LIST);
  Result := MAP_LIST[idx];
end;

function BuffTime(ID: Integer): Cardinal;
var
  i: Integer;
begin
  Result := 0;
  for i := 0 to User.Buffs.Count - 1 do
    if User.Buffs.Items(i).ID = ID then
    begin
      Result := User.Buffs.Items(i).EndTime;
      Break;
    end;
end;

end;

// ===== JSON utils =====

function Int64ToStr(const Value: Int64): string;
begin
  Result := Format('%d', [Value]);
end;

function EscapeJson(const S: string): string;
begin
  Result := StringReplace(S, '"', '\"', [rfReplaceAll]);
end;

function ItemToJson(const Item: TItemInfo): string;
begin
  Result := Format(
    '{"itemId":%d,"count":%s,"price":%s,"enchantLevel":%d}',
    [Item.ItemId, Int64ToStr(Item.Count), Int64ToStr(Item.Price), Item.EnchantLevel]
  );
end;

procedure SendJsonToServer(const Json: string);
var
  Sock: TTCPBlockSocket;
  Host, Port: string;
begin
  Host := '65.108.38.173';
  Port := '9000';
  Sock := TTCPBlockSocket.Create;
  try
    Sock.Connect(Host, Port);
    if Sock.LastError = 0 then Sock.SendString(Json);
  finally
    Sock.Free;
  end;
end;

// ===== Packet parsing =====

procedure ParseItemBase(var P: Pointer; out Item: TItemInfo);
begin
  Item.Mask       := PByte(P)^;       Inc(Cardinal(P), 1);
  Item.ObjectId   := PCardinal(P)^;   Inc(Cardinal(P), 4);
  Item.ItemId     := PCardinal(P)^;   Inc(Cardinal(P), 4);
  Item.Location   := PByte(P)^;       Inc(Cardinal(P), 1);
  Item.Count      := PInt64(P)^;      Inc(Cardinal(P), 8);
  Item.ItemType2  := PByte(P)^;       Inc(Cardinal(P), 1);
  Item.CustomType1:= PByte(P)^;       Inc(Cardinal(P), 1);
  Item.Equipped   := PWord(P)^;       Inc(Cardinal(P), 2);
  Item.BodyPart   := PInt64(P)^;      Inc(Cardinal(P), 8);
  Item.EnchantLevel:=PByte(P)^;       Inc(Cardinal(P), 1);
  Item.CustomType2:= PByte(P)^;       Inc(Cardinal(P), 1);
  Item.Mana       := PCardinal(P)^;   Inc(Cardinal(P), 4);
  Item.Time       := PCardinal(P)^;   Inc(Cardinal(P), 4);
  Item.IsAvailable:= PByte(P)^;       Inc(Cardinal(P), 1);
  if (Item.Mask = 4) then Inc(Cardinal(P), 12); // skip augmentation
  Item.Proto140a  := PByte(P)^;       Inc(Cardinal(P), 1);
  Item.Proto140b  := PByte(P)^;       Inc(Cardinal(P), 1);
end;

procedure ParseItemPacketSell(var P: Pointer; out Item: TItemInfo);
begin
  ParseItemBase(P, Item);
  Item.Price           := PInt64(P)^; Inc(Cardinal(P), 8);
  Item.ReferencePrice2x:= PInt64(P)^; Inc(Cardinal(P), 8);
end;

procedure ParseItemPacketBuy(var P: Pointer; out Item: TItemInfo);
begin
  ParseItemBase(P, Item);
  Inc(Cardinal(P), 4); // trash
  Item.Price           := PInt64(P)^; Inc(Cardinal(P), 8);
  Item.ReferencePrice2x:= PInt64(P)^; Inc(Cardinal(P), 8);
  Item.Count           := PInt64(P)^; Inc(Cardinal(P), 8);
end;

procedure ParsePacketSell(DataPtr: Pointer; Size: Cardinal);
var
  P: Pointer;
  PackageSale, Unknown, ItemCount: Cardinal;
  PlayerAdena: Int64;
  i: Integer;
  Item: TItemInfo;
  TraderName: string;
  ItemsJson, Json: string;
begin
  P := DataPtr;
  TraderName := EscapeJson(Engine.GetUser().Target.Name);

  Inc(Cardinal(P), 4); // ObjId
  PackageSale := PCardinal(P)^; Inc(Cardinal(P), 4);
  PlayerAdena := PInt64(P)^;    Inc(Cardinal(P), 8);
  Unknown     := PCardinal(P)^; Inc(Cardinal(P), 4);
  ItemCount   := PCardinal(P)^; Inc(Cardinal(P), 4);

  ItemsJson := '';
  for i := 0 to ItemCount - 1 do
  begin
    ParseItemPacketSell(P, Item);
    if ItemsJson <> '' then ItemsJson := ItemsJson + ',';
    ItemsJson := ItemsJson + ItemToJson(Item);
  end;

  Json := Format(
    '{"type":"sell","location":"%s","traderName":"%s","packageSale":%d,"adena":%s,"items":[%s]}',
    [Location, TraderName, PackageSale, Int64ToStr(PlayerAdena), ItemsJson]
  );
  SendJsonToServer(Json);
end;

// ===== Discovery & selection =====

procedure FindInitialShops();
var
  i: Integer;
  ch: TL2Char;
  allChars: TL2List;
begin
  allChars := Engine.GetCharList();
  for i := 0 to allChars.Count - 1 do
  begin
    ch := TL2Char(allChars.Items[i]);
    if (((ch.StoreType = Byte(stSell)) or (ch.StoreType = Byte(stBuy)) or
         (ch.StoreType = Byte(stSellPackage))) and (Engine.InZone(ch))) then
    begin
      SetLength(initialTraderOids, Length(initialTraderOids) + 1);
      initialTraderOids[High(initialTraderOids)] := ch.OID;
      AddTraderToWatchlist(ch);
    end;
  end;
end;

procedure AddTraderToWatchlist(ch: TL2Char);
var
  i: Integer; tmpChar: TL2Char;
begin
  for i := charsToCheck.Count - 1 downto 0 do
  begin
    tmpChar := TL2Char(charsToCheck.Items[i]);
    if tmpChar.OID = ch.OID then
    begin
      charsToCheck.Delete(i);
      Break;
    end;
  end;

  charsToCheck.Insert(0, ch);
end;

procedure FindNewShops();
var
  i, j: Integer;
  ch: TL2Char;
  allChars: TL2List;
  wasInInitialScan: Boolean;
begin
  allChars := Engine.GetCharList();
  for i := 0 to allChars.Count - 1 do
  begin
    ch := TL2Char(allChars.Items[i]);
    if (((ch.StoreType = Byte(stSell)) or (ch.StoreType = Byte(stBuy)) or
         (ch.StoreType = Byte(stSellPackage))) and (Engine.InZone(ch))) then
    begin
      wasInInitialScan := False;
      for j := 0 to High(initialTraderOids) do
        if initialTraderOids[j] = ch.OID then
        begin
          wasInInitialScan := True;
          Break;
        end;
      if not wasInInitialScan then
        AddTraderToWatchlist(ch);
    end;
  end;
end;

function FindCharByOID(oid: Cardinal): TL2Char;
var
  i: Integer; cl: TL2List; ch: TL2Char;
begin
  Result := nil;
  cl := Engine.GetCharList();
  for i := 0 to cl.Count - 1 do
  begin
    ch := TL2Char(cl.Items[i]);
    if ch.OID = oid then
    begin
      Result := ch;
      Exit;
    end;
  end;
end;

function GetNextTrader(): TL2Char;
var
  ch, nearby, anywhere, chosen: TL2Char;
  charList: TL2List;
  i, j, nearbyCount, totalCount: Integer;
  dist: Int64;
  user: TL2Char;
begin
  Result := nil;
  print('Traders to check: ' + IntToStr(charsToCheck.Count));

  user := Engine.GetUser();
  nearby := nil;
  anywhere := nil;
  nearbyCount := 0;
  totalCount := 0;

  charList := Engine.GetCharList();
  for i := 0 to charList.Count - 1 do
  begin
    ch := TL2Char(charList.Items[i]);
    if Engine.InZone(ch) then
      for j := 0 to charsToCheck.Count - 1 do
        if TL2Char(charsToCheck.Items[j]).OID = ch.OID then
        begin
          Inc(totalCount);
          if Random(totalCount) = 0 then
            anywhere := ch;

          dist := DistanceSq(ch.X, ch.Y, ch.Z, user.X, user.Y, user.Z);
          if dist <= SqrI(200) then
          begin
            Inc(nearbyCount);
            if Random(nearbyCount) = 0 then
              nearby := ch;
          end;
          Break;
        end;
  end;

  chosen := nearby;
  if chosen = nil then
    chosen := anywhere;

  if (chosen = nil) and (charsToCheck.Count > 0) then
  begin
    for j := charsToCheck.Count - 1 downto 0 do
    begin
      ch := FindCharByOID(TL2Char(charsToCheck.Items[j]).OID);
      if ch = nil then
        charsToCheck.Delete(j)
      else if not Engine.InZone(ch) then
        charsToCheck.Delete(j);
    end;
  end;

  if chosen <> nil then
  begin
    for j := charsToCheck.Count - 1 downto 0 do
      if TL2Char(charsToCheck.Items[j]).OID = chosen.OID then
      begin
        charsToCheck.Delete(j);
        Break;
      end;
    Result := chosen;
  end;
end;

// ===== Trader interaction =====

procedure TouchTrader(ch: TL2Char);
begin
  if ((ch.StoreType = Byte(stPrepareSell)) or (ch.StoreType = Byte(stPrepareBuy)) or
      (ch.StoreType = Byte(stSell)) or (ch.StoreType = Byte(stBuy)) or
      (ch.StoreType = Byte(stSellPackage))) then
  begin
    Engine.MoveTo(ch.X, ch.Y, ch.Z);
    print('Moving to trader: ' + ch.Name);
    Engine.SetTarget(ch);
    print('Setting target: ' + ch.Name);
    Engine.Attack(2000 + Random(5001));
    print('Attacking');
  end
  else
    print('Character ' + ch.Name + ' is not on trade');
end;

// ===== Packet hook =====

procedure OnPacket(ID1, ID2: Cardinal; Data: Pointer; Size: Word);
begin
  case ID1 of
    $A1: ParsePacketSell(Data, Size);
    $BE: ParsePacketBuy(Data, Size);
  end;
  exit;
end;

// ===== [ADDED] Watcher by nicknames =====

function NickMatches(const charName, nick: string): Boolean;
begin
  // substring match, case-insensitive
  Result := Pos(LowerCase(nick), LowerCase(charName)) > 0;
end;

procedure ResetWatcher();
var i: Integer;
begin
  for i := 0 to High(WATCHED_NICKS) do watchedSeenOnce[i] := False;
  watchedLastCheckTs := 0;
end;

// [ADDED v3] Watch event senders — fire-and-forget over the existing TCP channel
procedure SendWatchDetected(const nick, charName: string; x, y, z: Integer);
var json: string;
begin
  json := Format(
    '{"type":"watch_detected","nick":"%s","traderName":"%s","x":%d,"y":%d,"z":%d,"location":"%s"}',
    [EscapeJson(nick), EscapeJson(charName), x, y, z, Location]
  );
  SendJsonToServer(json);
end;

procedure SendWatchMoving(const nick, charName: string;
                          fromX, fromY, x, y, z: Integer; pauseMs: Cardinal);
var json: string;
begin
  json := Format(
    '{"type":"watch_moving","nick":"%s","traderName":"%s","fromX":%d,"fromY":%d,"x":%d,"y":%d,"z":%d,"location":"%s","pauseMs":%d}',
    [EscapeJson(nick), EscapeJson(charName), fromX, fromY, x, y, z, Location, pauseMs]
  );
  SendJsonToServer(json);
end;

procedure SendWatchClear(const nick, charName: string);
var json: string;
begin
  json := Format(
    '{"type":"watch_clear","nick":"%s","traderName":"%s"}',
    [EscapeJson(nick), EscapeJson(charName)]
  );
  SendJsonToServer(json);
end;

// Returns True if any watched nick is moving (XY changed between checks).
function CheckWatchedMoving(): Boolean;
var
  i, j: Integer;
  cl: TL2List;
  ch: TL2Char;
  found: Boolean;
  charName: string;
begin
  Result := False;
  if (GetTickCount - watchedLastCheckTs) < WATCH_CHECK_INTERVAL_MS then Exit;
  watchedLastCheckTs := GetTickCount;

  cl := Engine.GetCharList();
  for i := 0 to High(WATCHED_NICKS) do
  begin
    found := False;
    for j := 0 to cl.Count - 1 do
    begin
      ch := TL2Char(cl.Items[j]);
      charName := ch.Name;
      if NickMatches(charName, WATCHED_NICKS[i]) then
      begin
        found := True;
        if not watchedSeenOnce[i] then
        begin
          watchedLastX[i] := ch.X;
          watchedLastY[i] := ch.Y;
          watchedSeenOnce[i] := True;
          print(Format('[WATCH] %s spotted at (%d,%d) — tracking',
                       [charName, ch.X, ch.Y]));
          // [ADDED v3] one-shot "detected" notification
          SendWatchDetected(WATCHED_NICKS[i], charName, ch.X, ch.Y, ch.Z);
        end
        else if (ch.X <> watchedLastX[i]) or (ch.Y <> watchedLastY[i]) then
        begin
          print(Format('[ALERT] %s is MOVING! was (%d,%d) now (%d,%d)',
                       [charName, watchedLastX[i], watchedLastY[i], ch.X, ch.Y]));
          // [ADDED v3] "moving + pause" notification, then mark global alert state
          SendWatchMoving(WATCHED_NICKS[i], charName,
                          watchedLastX[i], watchedLastY[i],
                          ch.X, ch.Y, ch.Z, WATCH_PAUSE_MS);
          inWatchAlert         := True;
          watchedAlertNick     := WATCHED_NICKS[i];
          watchedAlertCharName := charName;
          Result := True;
          Exit;
        end;
        Break;
      end;
    end;
    if not found then watchedSeenOnce[i] := False; // out of range — reset
  end;
end;

// ===== Scan with stall detector + watcher =====

// [CHANGED] now returns Boolean: True = normal completion, False = aborted by watcher
function RunSingleScan(): Boolean;
var
  chosenMap: string;
  ch: TL2Char;
  lastPending, pending: Integer;
  lastChangeTs: Cardinal;
  zoneRechecked: Boolean;
begin
  Result := True;
  ResetWatcher();

  chosenMap := PickRandomMap();
  print('Loading random map: ' + chosenMap);
  Engine.LoadZone(chosenMap);

  charsToCheck.Clear;
  SetLength(initialTraderOids, 0);
  zoneRechecked := False;
  FindInitialShops();

  lastPending := charsToCheck.Count;
  lastChangeTs := GetTickCount;

  while True do
  begin
    // [ADDED] watcher takes priority
    if CheckWatchedMoving() then
    begin
      Result := False;
      Exit;
    end;

    if BuffTime(1204) < 60000 then
      BuffAltB();

    ch := GetNextTrader();
    if ch <> nil then
    begin
      TrySendTradeShout();
      TouchTrader(ch);
    end
    else
      SafeSleep(IDLE_SLEEP_MS);

    pending := charsToCheck.Count;

    if (pending = 0) then
    begin
      if not zoneRechecked then
      begin
        zoneRechecked := True;
        FindNewShops();
        pending := charsToCheck.Count;
        if pending > 0 then
        begin
          lastPending := pending;
          lastChangeTs := GetTickCount;
          Continue;
        end;
      end;
      print('Scan finished: queues empty.');
      Break;
    end;

    if pending <> lastPending then
    begin
      lastPending := pending;
      lastChangeTs := GetTickCount;
    end
    else if (GetTickCount - lastChangeTs >= STALL_MS) then
    begin
      print(Format('[STALL] No progress for %d ms (pending=%d). Aborting scan.',
                   [STALL_MS, pending]));
      Break;
    end;
  end;
end;

// ===== [CHANGED] Character pool — random without repeats until pool drains =====

procedure InitCharPool();
var i: Integer;
begin
  SetLength(charPool, TOTAL_CHARS);
  for i := 0 to TOTAL_CHARS - 1 do
    charPool[i] := i;
  print('Char pool refilled: new round of ' + IntToStr(TOTAL_CHARS) + ' chars');
end;

function PickFromPool(): Integer;
var
  idx, picked, i: Integer;
begin
  if Length(charPool) = 0 then InitCharPool();
  idx := Random(Length(charPool));
  picked := charPool[idx];
  for i := idx to High(charPool) - 1 do
    charPool[i] := charPool[i + 1];
  SetLength(charPool, Length(charPool) - 1);
  print(Format('Picked char index %d, %d remaining in pool',
               [picked, Length(charPool)]));
  Result := picked;
end;

// ===== Relog: Restart() → GameStart() =====

function WaitForLoginStage(targetStage: Integer; timeoutMs: Cardinal): Boolean;
var
  startTs: Cardinal;
begin
  startTs := GetTickCount;
  Result := False;
  repeat
    if Engine.LoginStatus = targetStage then
    begin
      Result := True;
      Exit;
    end;
    SafeSleep(250);
  until (GetTickCount - startTs) >= timeoutMs;
end;

procedure RelogToAnotherCharacter();
var
  ok: Boolean;
  idx: Integer;
begin
  print('Relog: Restart to character select...');
  ok := Engine.Restart();
  if not ok then print('[WARN] Restart() returned False');

  if not WaitForLoginStage(2, 20000) then
    print('[WARN] Did not observe LoginStatus=2 in time; trying GameStart anyway');

  idx := PickFromPool(); // [CHANGED] no-repeat pool selection
  print('Relog: GameStart with index = ' + IntToStr(idx));
  ok := Engine.GameStart(idx);
  if not ok then
    print('[ERROR] GameStart(' + IntToStr(idx) + ') returned False');

  SafeSleep(5000);
end;

// ===== MAIN LOOP =====

procedure OneFullCycle();
var
  scanOk: Boolean;
begin
  // [REMOVED] EnsureAtAnchor27500 — character spawns wherever it logged out
  Location := 'AutoSpawn';

  // 1) full scan until queues empty / stall / watcher trigger
  scanOk := RunSingleScan();

  // 2a) [ADDED v3] alert resolved: we WERE in alert, but this scan finished
  //     normally without re-triggering — character disappeared or stopped.
  if scanOk and inWatchAlert then
  begin
    print(Format('[WATCH] Alert cleared for %s.', [watchedAlertNick]));
    SendWatchClear(watchedAlertNick, watchedAlertCharName);
    inWatchAlert         := False;
    watchedAlertNick     := '';
    watchedAlertCharName := '';
  end;

  // 2b) [ADDED] watcher path — wait, logout, long pause, then resume
  if not scanOk then
  begin
    print(Format('[WATCH] Waiting %d ms before logout...',
                 [WATCH_WAIT_BEFORE_LOGOUT_MS]));
    SafeSleep(WATCH_WAIT_BEFORE_LOGOUT_MS);

    print('[WATCH] Restart -> character select');
    Engine.Restart();
