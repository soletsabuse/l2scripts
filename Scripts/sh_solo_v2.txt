uses SysUtils;
uses NecroFightV2 in 'necro_fight_v2.pas';

// Necro solo: necromancer with a summon. Only settings here, all logic is in necro_fight_v2.pas.
// Fight + rebuff at the safe point + supplies + anti-disarm + patrol.

// the bot sends game events only to the script itself - pass them on to the library
// (needed for "Can't see target."); do not remove
procedure OnAction(Action: TL2Action; P1, P2: cardinal);
begin
  NecroOnAction(Action, P1, P2);
end;

begin
  // bot config loaded at start (file in Library\Configurations, e.g. only event notifications,
  // no fighting - the script fights itself); remove the line = no config
  SetConfig('alarm.xml');

  // ===== our skills, PRIORITY order inside each kind =====
  // AddMySkill(skill ID, kind, param)
  //   KIND_DRAIN  - param = HP %: spammed while our HP is below it
  //   KIND_DEBUFF - param = ms: cast again on the SAME mob only after this
  //   KIND_NUKE   - param unused: always spammed (the one that reloads first)
  // AddMyAoe(skill ID, radius, cast range) - AoE: only when ready, 3+ mobs within radius of the target
  //   and the target closer than cast range (0 = any distance)
  // AddMyDebuff(skill ID, recast ms, min mob HP) - a debuff cast only on mobs with more HP than that
  AddMySkill(1234, KIND_DRAIN,  85);      // Vampiric Claw, HP < 70 %
  AddMySkill(1343, KIND_DRAIN,  60);       // Vortex Dark
  AddMySkill(1074, KIND_DEBUFF, 6000);   // debuff, again on the same mob after 30 sec
  
  AddMySkill(1341, KIND_NUKE,  6500);       // Wind Vortex
  AddMySkill(1239, KIND_NUKE,   0);       // Death Spike
  AddMyAoe(1176, 300, 500);   // AoE: 3+ mobs within 400 of the target, target closer than 600 (set yours)
  // AddMyPbAoe(skill ID, radius, min mobs) - AoE around US: only when ready and at least
  // min mobs are within radius of us (put your skill ID and its radius)
  AddMyPbAoe(1417, 200, 3);

  AddMyDebuff(1074, 13000, 100000); 
  // how often a skill packet is sent (ms)
  SetSpamMs(100);
  // a player (not party / clan / AddNoAlertPlayer) hits us -> after a random 0.7-1.5 sec our drain / nukes
  // at him until he stops / dies / leaves; no line = off
  SetPvpAnswer(700, 1500);

  // ===== summon =====
  //SetSummon(1154);   // summon skill (summoned when missing)

  // summon skills = action IDs: AddPetSkill(action, kind, param)
  //   KIND_DEBUFF - again on the same mob after param ms
  //   KIND_NUKE   - every param ms
  //   KIND_DRAIN  - when summon HP % < param
  //AddPetSkill(1200, KIND_DEBUFF, 8000);   // M.Def down on the mob
  //AddPetSkill(47,   KIND_NUKE,   5000);   // damage

  // summon attack order (runs in, hits with hands): -1 = never, 0 = every mob, N = only mobs with MaxHP >= N
  //SetPetAttack(-1);

  // ===== targets: only inside the zone active in the bot (no zone -> radius around the start) =====
  SetTargetRange(800);   // new targets / current target only this close
  SetAggroRange(800);    // mobs on us / the summon only this close
  SetFreeRadius(800);    // no zone in the bot: farm this close to the start point
  SetCastRange(600);     // our skills reach this far: a target farther away -> walk closer first
  SetHeightRange(400);   // mobs higher / lower than us by more than this are never targeted (0 = any)

  // a mob that just respawned is taken (target / pull / walk to it) only after a random 0.5-1 sec
  // (500..1000 ms); mobs that hit us are fought at once. No line = at once
  SetNewMobDelay(500, 1000);

  // ===== pack gathering: SetPull(range, min mobs, max mobs, tap skill) =====
  //   nobody on us and at least MIN free mobs within RANGE -> every mob there that is NOT aggroed
  //   gets a nuke from SetPullTapRange (again if it did not aggro); only mobs within RANGE of the point
  //   where it started - no running far away. MAX mobs on us (MIN if MAX = 0) / nothing free left /
  //   20 sec -> attack the pack with the AoE. Fewer than MIN around -> no chasing single mobs,
  //   go to the next group of MIN+.
  //   TAP skill = the nuke for those hits (0 = our nuke that is ready first)
  SetPull(1000, 4, 8, 1239);
  // the tap goes out from this far (the real range of the TAP skill; no line = SetCastRange): we stand
  // back and the mobs come to us. A tapped mob that did not aggro (the skill reaches less, a wall) is
  // tapped again 150 closer each time, down to SetCastRange. The tap skill in reuse -> we wait standing
  SetPullTapRange(900);

  // extra names to never target (other players' summons, chests... are skipped by default)
  // AddIgnoreName('Some Mob');
  AddIgnoreID(32049);   // NPC ID 32049 - never targeted

  // champions (MaxHP above SetChampionHP): true = never taken as a new target, not run to, not pulled,
  // not hit by our AoE - fought only when the champion is on us / the summon itself
  SetSkipChampions(true);
  // SetChampionHP(100000);   // a mob with MaxHP above this is a champion (the champion alert too)

  // ===== "player close" alert of the script: 0 = off (the profile alarm.xml watches players);
  // the champion alert stays on. SetPlayerAlert(700) - back on; the nicks below never trigger it
  // (and are never answered by SetPvpAnswer) =====
  SetPlayerAlert(0);
  AddNoAlertPlayer('kkk');
  AddNoAlertPlayer('eee');
  AddNoAlertPlayer('elephantKING');
  AddNoAlertPlayer('antKING');
  AddNoAlertPlayer('horseKING');
  // AddNoAlertPlayer('Friend');

  // ===== rebuff: watched buff ID + buff bypasses =====
  // buff running out -> fight off until OUT of combat -> safe point (if set) -> summon ->
  // buff (only out of combat) -> supplies
  SetBuff(1204, [
    '_bbs_buffer_show',
    'book_use?name=s&pre=0&target=Player'
  ]);
  // extra bypasses for the summon, sent right after ours (no line = the summon gets buffed with us)
  // SetSummonBuff(1204, ['book_use?...']);   // buff watched on the summon (0 = same as ours) + extra bypasses

  // where to rebuff (no line = right where we are)
  //SetSafePoint(-20364, -53432, -11000);

  // ===== patrol: no targets -> run these points one by one =====
  // attacked / a mob shows up on the way -> fight it, then run on to the same point

  // with patrol, new mobs only this close (mobs that attack us are always fought, up to 900)
  SetPatrolRange(200);

  // ===== anti-disarm: the weapon in hands at start is remembered (or set it here) =====
  // SetWeapon(1234);

  // ===== supplies: AddSupply(ITEM, min, max) - less than min -> buy up to max =====
  AddSupply(BLESSED_SPIRITSHOT_B,     5000, 5000);
  AddSupply(BEAST_BLESSED_SPIRITSHOT, 5000, 5000);
  AddSupply(SPIRIT_ORE,              10000, 20000);
  AddSupply(MANA_DRUG,                300, 500);

  RunNecro();
end.
