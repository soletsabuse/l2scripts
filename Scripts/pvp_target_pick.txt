uses SysUtils;

// =============================================================================
// PVP target picker -- runs as a daemon, polls a hotkey, picks the enemy whose
// direction puts the most hostile players into a 90deg fan cone, SetTarget's
// them.
//
// Picking strategy:
//   for each candidate primary T (alive, in hostile clan, matching class,
//   within SKILL_RANGE, within HEIGHT_RANGE Z):
//     direction := normalize(T - me)
//     count := number of valid enemies whose angle from `direction`
//             (measured at me) is <= FAN_HALF_ANGLE_DEG
//   pick T with highest count, SetTarget(T).
// =============================================================================

// =============================================================================
// CONFIG
// =============================================================================
var
  // Hotkey -- VK code. Default: Space ($20).
  // VK codes:
  //   Space=$20  Tab=$09  Enter=$0D  Esc=$1B  Backspace=$08
  //   F1=$70 F2=$71 ... F8=$77 F12=$7B
  //   'A'..'Z' = $41..$5A   '0'..'9' = $30..$39
  HOTKEY_VK          : integer = $20;       // Space

  // Hostile clan names. Exact case-sensitive match. [''] = match any clan.
  ENEMY_CLANS        : array of string = ['PLAYHARDGOPRO', 'GhostRiders', 'DeathSentence', 'HeavenSent', 'StayGrounded', 'Slayers'];

  // Class IDs to focus on. Mage 3rd-profession on most L2 Classic servers:
  //   16 = Bishop
  //   27 = Spellsinger ("Mystic Muse" on some servers)
  //   13 = Necromancer
  //   36 = Spellhowler
  //   39 = Shillien Elder
  // If your server uses awakened class IDs, try instead:
  //   97=Cardinal, 95=Mystic Muse, 99=Soultaker, 96=Storm Screamer, 100=Shillien Saint
  ENEMY_CLASS_IDS    : array of integer = [112, 110, 105, 103, 98, 97, 95, 94];

  // Skill geometry, mirrors s_multiple_shot1:
  //   fan_range = {0;0;700;90}     -> 700u radius, 90deg total cone
  //   affect_scope_height = {-200;200}
  SKILL_RANGE        : integer = 700;
  FAN_HALF_ANGLE_DEG : integer = 45;        // half of 90, in degrees
  HEIGHT_RANGE       : integer = 200;       // |dz| limit

  // Polling interval in ms. Lower = more responsive but burns more CPU.
  KEY_POLL_MS        : integer = 30;

  // Raw hex packet bytes sent after SetTarget to mark the target.
  // Sniffed on this server with token_sniffer.txt (manual mark):
  //   56 = RequestActionUse, action 0x50 (=80) LE, ctrl 0 (4 bytes), shift 0 (1 byte)
  // Other marks are other action IDs - sniff them the same way if needed.
  // (Old chronicles used 'B36F000000' = RequestUserCommand /usetoken3 - does NOT work here.)
  // Set to '' to skip the mark.
  TOKEN_PACKET_HEX   : string  = '56500000000000000000';

// =============================================================================
// WinAPI -- read keyboard async-state directly. Returns smallint where the
// high-bit (0x8000) is set when the key is currently pressed.
//   Standard Delphi external-declaration syntax. If this PaxScript dialect
//   doesn't support DLL imports, this is the line that'll fail to compile --
//   tell me and we'll find an alternative.
// =============================================================================
function GetAsyncKeyState(vKey: integer): smallint; stdcall; external 'user32.dll';

// =============================================================================
// Math helpers
// =============================================================================
const
  PI_VAL = 3.14159265358979;

// Clan is one of ENEMY_CLANS. An empty name in the list ('') = any clan.
function IsEnemyClan(Clan: string): boolean;
var c: integer;
begin
  result := false;
  for c := Low(ENEMY_CLANS) to High(ENEMY_CLANS) do
    if (ENEMY_CLANS[c] = '') or (Clan = ENEMY_CLANS[c]) then begin
      result := true;
      exit;
    end;
end;

// clans as one line for the start message
function ClansText(): string;
var c: integer;
begin
  result := '';
  for c := Low(ENEMY_CLANS) to High(ENEMY_CLANS) do begin
    if result <> '' then result := result + ', ';
    result := result + '"' + ENEMY_CLANS[c] + '"';
  end;
end;

// Returns true if CharList(i) is a valid candidate. If IgnoreClass=true the
// class filter is skipped (clan and other filters still apply).
function IsValidEnemyEx(i: integer; IgnoreClass: boolean): boolean;
var
  PX, PY, PZ: integer;
  c, cls: integer;
  ClassOK: boolean;
begin
  result := false;
  if CharList(i).Dead then exit;
  if CharList(i).OID = User.OID then exit;          // never self

  if not IsEnemyClan(CharList(i).Clan) then exit;

  if (not IgnoreClass) and (Length(ENEMY_CLASS_IDS) > 0) then begin
    ClassOK := false;
    cls := CharList(i).ClassID;
    for c := Low(ENEMY_CLASS_IDS) to High(ENEMY_CLASS_IDS) do
      if cls = ENEMY_CLASS_IDS[c] then begin
        ClassOK := true;
        break;
      end;
    if not ClassOK then exit;
  end;

  PZ := CharList(i).Z;
  if Abs(PZ - User.Z) > HEIGHT_RANGE then exit;

  PX := CharList(i).X;
  PY := CharList(i).Y;
  if User.DistTo(PX, PY, PZ) > SKILL_RANGE then exit;

  result := true;
end;

// Count valid enemies inside a 2D fan from (User pos) aimed at (TX, TY).
// Includes the primary itself (it's at angle 0). IgnoreClass forwards to the
// validity check.
function CountInFanEx(TX, TY: integer; CosHalfFan: double; IgnoreClass: boolean): integer;
var
  i: integer;
  UX, UY: integer;
  DX, DY, DLen: double;
  EX, EY: integer;
  EVecX, EVecY, ELen: double;
  Dot, CosAng: double;
begin
  result := 0;
  UX := User.X;
  UY := User.Y;

  DX := TX - UX;
  DY := TY - UY;
  DLen := Sqrt(DX*DX + DY*DY);
  if DLen < 1 then exit;     // primary on top of us -- can't aim
  DX := DX / DLen;
  DY := DY / DLen;

  for i := 0 to CharList.Count - 1 do begin
    if not IsValidEnemyEx(i, IgnoreClass) then continue;
    EX := CharList(i).X;
    EY := CharList(i).Y;
    EVecX := EX - UX;
    EVecY := EY - UY;
    ELen := Sqrt(EVecX*EVecX + EVecY*EVecY);
    if ELen < 1 then continue;
    Dot := DX * EVecX + DY * EVecY;
    CosAng := Dot / ELen;
    if CosAng >= CosHalfFan then
      result := result + 1;
  end;
end;

// Find the primary target whose facing-fan covers the most valid enemies.
// Returns -1 if no candidates exist.
function FindBestPrimary(CosHalfFan: double; IgnoreClass: boolean): integer;
var
  i, BestIdx, BestCnt, Cnt: integer;
  TX, TY: integer;
begin
  BestIdx := -1;
  BestCnt := 0;
  for i := 0 to CharList.Count - 1 do begin
    if not IsValidEnemyEx(i, IgnoreClass) then continue;
    TX := CharList(i).X;
    TY := CharList(i).Y;
    Cnt := CountInFanEx(TX, TY, CosHalfFan, IgnoreClass);
    if Cnt > BestCnt then begin
      BestCnt := Cnt;
      BestIdx := i;
    end;
  end;
  result := BestIdx;
end;

// Send the token-mark packet directly. Bypasses the chat UI -- we just push
// the same bytes the client would send if you typed /usetoken3 yourself.
procedure SendTokenPacket();
begin
  if TOKEN_PACKET_HEX = '' then exit;
  Engine.SendToServer(TOKEN_PACKET_HEX);
end;

// =============================================================================
// PICK -- find best primary (with class filter, then any class), SetTarget,
// drop the moon mark on them via /usetoken3.
// =============================================================================
procedure PickPvPTarget();
var
  BestIdx, BestCnt: integer;
  TX, TY: integer;
  OID: integer;
  CosHalfFan: double;
  UsedFallback: boolean;
begin
  CosHalfFan := Cos(FAN_HALF_ANGLE_DEG * PI_VAL / 180.0);
  UsedFallback := false;

  // first pass: with class filter
  BestIdx := FindBestPrimary(CosHalfFan, false);

  // fallback: no targets in the wanted classes -> drop class filter, keep clan
  if BestIdx < 0 then begin
    BestIdx := FindBestPrimary(CosHalfFan, true);
    UsedFallback := true;
  end;

  if BestIdx < 0 then begin
    Engine.Msg('[Pvp]', 'No valid enemy in range', 128);
    exit;
  end;

  // recompute fan coverage on the chosen primary for log
  TX := CharList(BestIdx).X;
  TY := CharList(BestIdx).Y;
  BestCnt := CountInFanEx(TX, TY, CosHalfFan, UsedFallback);

  OID := CharList(BestIdx).OID;
  Engine.SetTarget(OID);

  // Small gap so the server registers the new target before the token packet
  // arrives (otherwise the mark may land on the previous target).
  Delay(50);

  // Drop the moon mark (/usetoken3) directly via raw packet -- visible to
  // allies/party for focus call.
  SendTokenPacket;

  Engine.Msg('[Pvp]',
    'Target: ' + CharList(BestIdx).Name +
    ' OID=' + IntToStr(OID) +
    ' (fan covers ' + IntToStr(BestCnt) + ' enemies)' +
    ' [' + IntToStr(Ord(UsedFallback)) + '=fallback]', 128);
end;

// =============================================================================
// Hotkey watcher loop. Edge-triggered: fires on press, not while held.
// =============================================================================
var
  KeyWasDown: boolean = false;
  KeyDown:    boolean;

begin
  Engine.Msg('[Pvp]',
    'Started -- hotkey VK=$' + IntToHex(HOTKEY_VK, 2) +
    ' clans=' + ClansText() +
    ' classes=' + IntToStr(Length(ENEMY_CLASS_IDS)) +
    ' range=' + IntToStr(SKILL_RANGE) +
    ' fan=' + IntToStr(FAN_HALF_ANGLE_DEG * 2) + 'deg', 128);

  while Delay(KEY_POLL_MS) do begin
    KeyDown := (GetAsyncKeyState(HOTKEY_VK) and $8000) <> 0;
    if KeyDown then begin
      if not KeyWasDown then begin
        PickPvPTarget;
        KeyWasDown := true;
      end;
    end else
      KeyWasDown := false;
  end;
end.
