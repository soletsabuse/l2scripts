uses SysUtils;

// =============================================================================
// PVP target picker -- runs as a daemon, polls hotkeys, takes a target and puts
// a mark (token) on it. Only players from ENEMY_CLANS are ever picked.
//
//   Space - fan mode: the enemy whose direction puts the most enemies into a
//           90deg cone (multiple shot), classes from FAN_CLASS_IDS
//   Z     - nearest Mystic Muse / Soultaker (Spellsinger / Necromancer)
//   X     - nearest Cardinal / Eva's Saint (Bishop / Elven Elder)
//   C     - res-kill: nearest enemy that was resurrected in the last RESKILL_SEC
// =============================================================================

// =============================================================================
// CONFIG
// =============================================================================
var
  // Hotkeys -- VK codes:
  //   Space=$20  Tab=$09  Enter=$0D  Esc=$1B  Backspace=$08
  //   F1=$70 F2=$71 ... F8=$77 F12=$7B
  //   'A'..'Z' = $41..$5A   '0'..'9' = $30..$39
  KEY_FAN            : integer = $20;       // Space
  KEY_MAGES          : integer = $5A;       // Z
  KEY_HEALERS        : integer = $58;       // X
  KEY_RESKILL        : integer = $43;       // C

  // Hostile clan names. Exact case-sensitive match. [''] = match any clan.
  ENEMY_CLANS        : array of string = ['PLAYHARDGOPRO', 'GhostRiders', 'DeathSentence', 'HeavenSent', 'StayGrounded', 'Slayers'];

  // Z: Mystic Muse 103 (Spellsinger 27), Soultaker 95 (Necromancer 13)
  MAGE_CLASS_IDS     : array of integer = [103, 27, 95, 13];
  // X: Cardinal 97 (Bishop 16), Eva's Saint 105 (Elven Elder 30)
  HEALER_CLASS_IDS   : array of integer = [97, 16, 105, 30];
  // Space (fan mode): classes to aim at first; no one of them -> any class of the enemy clans
  FAN_CLASS_IDS      : array of integer = [112, 110, 105, 103, 98, 97, 95, 94];

  // Z / X / C: pick only this close
  PICK_RANGE         : integer = 1500;
  // C: "just resurrected" = alive again not longer than this ago
  RESKILL_SEC        : integer = 15;

  // Fan geometry (Space), mirrors s_multiple_shot1:
  //   fan_range = {0;0;700;90}     -> 700u radius, 90deg total cone
  //   affect_scope_height = {-200;200}
  SKILL_RANGE        : integer = 700;
  FAN_HALF_ANGLE_DEG : integer = 45;        // half of 90, in degrees
  HEIGHT_RANGE       : integer = 200;       // |dz| limit (all modes)

  // Polling interval in ms. Lower = more responsive but burns more CPU.
  KEY_POLL_MS        : integer = 30;

  // Raw hex packet bytes sent after SetTarget to mark the target.
  // Sniffed on this server with token_sniffer.txt (manual mark):
  //   56 = RequestActionUse, action 0x50 (=80) LE, ctrl 0 (4 bytes), shift 0 (1 byte)
  // Other marks are other action IDs - sniff them the same way if needed.
  // Set to '' to skip the mark.
  TOKEN_PACKET_HEX   : string  = '56500000000000000000';

// =============================================================================
// WinAPI -- read keyboard async-state directly. Returns smallint where the
// high-bit (0x8000) is set when the key is currently pressed.
// =============================================================================
function GetAsyncKeyState(vKey: integer): smallint; stdcall; external 'user32.dll';

const
  PI_VAL = 3.14159265358979;
  REVIVE_SLOTS = 64;

// =============================================================================
// Resurrection tracking (C): we remember who was dead; dead -> alive = revived now
// =============================================================================
var
  DeadOID: array[0..REVIVE_SLOTS - 1] of integer;      // players seen dead
  ReviveOID: array[0..REVIVE_SLOTS - 1] of integer;    // players seen getting up
  ReviveTime: array[0..REVIVE_SLOTS - 1] of cardinal;
  DeadHead, ReviveHead: integer;
  LastReviveScan: cardinal;

function WasDead(OID: integer): boolean;
var k: integer;
begin
  result := false;
  for k := 0 to REVIVE_SLOTS - 1 do
    if DeadOID[k] = OID then begin result := true; exit; end;
end;

procedure ForgetDead(OID: integer);
var k: integer;
begin
  for k := 0 to REVIVE_SLOTS - 1 do
    if DeadOID[k] = OID then DeadOID[k] := 0;
end;

procedure RememberDead(OID: integer);
begin
  if WasDead(OID) then exit;
  DeadOID[DeadHead] := OID;
  DeadHead := (DeadHead + 1) mod REVIVE_SLOTS;
end;

procedure RememberRevive(OID: integer);
begin
  ReviveOID[ReviveHead] := OID;
  ReviveTime[ReviveHead] := GetTickCount;
  ReviveHead := (ReviveHead + 1) mod REVIVE_SLOTS;
end;

// revived not longer than RESKILL_SEC ago
function JustRevived(OID: integer): boolean;
var k: integer;
begin
  result := false;
  for k := 0 to REVIVE_SLOTS - 1 do
    if (ReviveOID[k] = OID) and (GetTickCount - ReviveTime[k] <= cardinal(RESKILL_SEC * 1000)) then begin
      result := true;
      exit;
    end;
end;

// every ~200 ms: who is dead now, who got up since the last look
procedure ScanRevives();
var i, OID: integer;
begin
  if GetTickCount - LastReviveScan < 200 then exit;
  LastReviveScan := GetTickCount;
  for i := 0 to CharList.Count - 1 do begin
    OID := CharList(i).OID;
    if CharList(i).Dead then
      RememberDead(OID)
    else if WasDead(OID) then begin
      ForgetDead(OID);
      RememberRevive(OID);
    end;
  end;
end;

// =============================================================================
// Filters
// =============================================================================

// Clan is one of ENEMY_CLANS. An empty name in the list ('') = any clan.
function IsEnemyClan(Clan: string): boolean;
var c: integer;
begin
  result := false;
  for c := Low(ENEMY_CLANS) to High(ENEMY_CLANS) do
    if (ENEMY_CLANS[c] = '') or (Clan = ENEMY_CLANS[c]) then begin
      result := true;
      exit;
    end;
end;

function ClassInList(cls: integer; List: array of integer): boolean;
var c: integer;
begin
  result := false;
  for c := Low(List) to High(List) do
    if cls = List[c] then begin
      result := true;
      exit;
    end;
end;

// alive enemy-clan player (not us) within Range and HEIGHT_RANGE
function IsEnemyInRange(i, Range: integer): boolean;
var PX, PY, PZ: integer;
begin
  result := false;
  if CharList(i).Dead then exit;
  if CharList(i).OID = User.OID then exit;
  if not IsEnemyClan(CharList(i).Clan) then exit;
  PX := CharList(i).X;
  PY := CharList(i).Y;
  PZ := CharList(i).Z;
  if Abs(PZ - User.Z) > HEIGHT_RANGE then exit;
  result := User.DistTo(PX, PY, PZ) <= Range;
end;

// clans as one line for the start message
function ClansText(): string;
var c: integer;
begin
  result := '';
  for c := Low(ENEMY_CLANS) to High(ENEMY_CLANS) do begin
    if result <> '' then result := result + ', ';
    result := result + '"' + ENEMY_CLANS[c] + '"';
  end;
end;

// =============================================================================
// Target + mark
// =============================================================================

procedure SendTokenPacket();
begin
  if TOKEN_PACKET_HEX = '' then exit;
  Engine.SendToServer(TOKEN_PACKET_HEX);
end;

procedure TakeTarget(i: integer; Mode: string);
var OID: integer;
begin
  OID := CharList(i).OID;
  Engine.SetTarget(OID);
  Delay(50);   // let the server register the target before the mark
  SendTokenPacket;
  Engine.Msg('[Pvp]', Mode + ': ' + CharList(i).Name + ' [' + CharList(i).Clan + ']', 128);
end;

// =============================================================================
// Z / X: nearest enemy of the given classes
// =============================================================================
procedure PickNearestOfClasses(Classes: array of integer; Mode: string);
var i, Best, PX, PY, PZ: integer;
  D, BestD: double;
begin
  Best := -1;
  BestD := 999999;
  for i := 0 to CharList.Count - 1 do begin
    if not IsEnemyInRange(i, PICK_RANGE) then continue;
    if not ClassInList(CharList(i).ClassID, Classes) then continue;
    PX := CharList(i).X;
    PY := CharList(i).Y;
    PZ := CharList(i).Z;
    D := User.DistTo(PX, PY, PZ);
    if D < BestD then begin
      BestD := D;
      Best := i;
    end;
  end;
  if Best < 0 then
    Engine.Msg('[Pvp]', Mode + ': nobody in range', 128)
  else
    TakeTarget(Best, Mode);
end;

// =============================================================================
// C: res-kill - nearest enemy that just got up
// =============================================================================
procedure PickResKill();
var i, Best, PX, PY, PZ: integer;
  D, BestD: double;
begin
  ScanRevives();
  Best := -1;
  BestD := 999999;
  for i := 0 to CharList.Count - 1 do begin
    if not IsEnemyInRange(i, PICK_RANGE) then continue;
    if not JustRevived(CharList(i).OID) then continue;
    PX := CharList(i).X;
    PY := CharList(i).Y;
    PZ := CharList(i).Z;
    D := User.DistTo(PX, PY, PZ);
    if D < BestD then begin
      BestD := D;
      Best := i;
    end;
  end;
  if Best < 0 then
    Engine.Msg('[Pvp]', 'Res-kill: nobody resurrected in the last ' + IntToStr(RESKILL_SEC) + ' sec', 128)
  else
    TakeTarget(Best, 'Res-kill');
end;

// =============================================================================
// Space: fan mode (the enemy whose direction covers the most enemies)
// =============================================================================

function IsValidFanEnemy(i: integer; IgnoreClass: boolean): boolean;
begin
  result := IsEnemyInRange(i, SKILL_RANGE);
  if result and (not IgnoreClass) and (Length(FAN_CLASS_IDS) > 0) then
    result := ClassInList(CharList(i).ClassID, FAN_CLASS_IDS);
end;

// Count valid enemies inside a 2D fan from (User pos) aimed at (TX, TY).
function CountInFanEx(TX, TY: integer; CosHalfFan: double; IgnoreClass: boolean): integer;
var
  i, UX, UY, EX, EY: integer;
  DX, DY, DLen, EVecX, EVecY, ELen, Dot: double;
begin
  result := 0;
  UX := User.X;
  UY := User.Y;
  DX := TX - UX;
  DY := TY - UY;
  DLen := Sqrt(DX*DX + DY*DY);
  if DLen < 1 then exit;
  DX := DX / DLen;
  DY := DY / DLen;
  for i := 0 to CharList.Count - 1 do begin
    if not IsValidFanEnemy(i, IgnoreClass) then continue;
    EX := CharList(i).X;
    EY := CharList(i).Y;
    EVecX := EX - UX;
    EVecY := EY - UY;
    ELen := Sqrt(EVecX*EVecX + EVecY*EVecY);
    if ELen < 1 then continue;
    Dot := DX * EVecX + DY * EVecY;
    if Dot / ELen >= CosHalfFan then result := result + 1;
  end;
end;

function FindBestPrimary(CosHalfFan: double; IgnoreClass: boolean): integer;
var i, BestCnt, Cnt, TX, TY: integer;
begin
  result := -1;
  BestCnt := 0;
  for i := 0 to CharList.Count - 1 do begin
    if not IsValidFanEnemy(i, IgnoreClass) then continue;
    TX := CharList(i).X;
    TY := CharList(i).Y;
    Cnt := CountInFanEx(TX, TY, CosHalfFan, IgnoreClass);
    if Cnt > BestCnt then begin
      BestCnt := Cnt;
      result := i;
    end;
  end;
end;

procedure PickFanTarget();
var Best: integer;
  CosHalfFan: double;
begin
  CosHalfFan := Cos(FAN_HALF_ANGLE_DEG * PI_VAL / 180.0);
  Best := FindBestPrimary(CosHalfFan, false);
  if Best < 0 then Best := FindBestPrimary(CosHalfFan, true);   // no wanted class -> any class of the enemy clans
  if Best < 0 then
    Engine.Msg('[Pvp]', 'Fan: no valid enemy in range', 128)
  else
    TakeTarget(Best, 'Fan');
end;

// =============================================================================
// Hotkey watcher loop. Edge-triggered: fires on press, not while held.
// =============================================================================
var
  FanWas, MagesWas, HealersWas, ResWas: boolean;

function Pressed(VK: integer): boolean;
begin
  result := (GetAsyncKeyState(VK) and $8000) <> 0;
end;

begin
  Engine.Msg('[Pvp]', 'Started -- Space = fan, Z = Mystic Muse / Soultaker, X = Cardinal / Eva''s Saint, ' +
    'C = res-kill; clans ' + ClansText(), 128);

  while Delay(KEY_POLL_MS) do begin
    ScanRevives();   // keep the dead / revived list fresh for C

    if Pressed(KEY_FAN) then begin
      if not FanWas then PickFanTarget;
      FanWas := true;
    end else FanWas := false;

    if Pressed(KEY_MAGES) then begin
      if not MagesWas then PickNearestOfClasses(MAGE_CLASS_IDS, 'Mystic Muse / Soultaker');
      MagesWas := true;
    end else MagesWas := false;

    if Pressed(KEY_HEALERS) then begin
      if not HealersWas then PickNearestOfClasses(HEALER_CLASS_IDS, 'Cardinal / Eva''s Saint');
      HealersWas := true;
    end else HealersWas := false;

    if Pressed(KEY_RESKILL) then begin
      if not ResWas then PickResKill;
      ResWas := true;
    end else ResWas := false;
  end;
end.
